   701→    } catch {
   702→      // Detect top-level structure to prefer the right regex
   703→      const trimmed = content.trim()
   704→      let jsonMatch: string | null = null
   705→      if (trimmed.startsWith('{')) {
   706→        jsonMatch = (content.match(/\{[\s\S]*\}/) || [])[0] || null

   715→          parsed = JSON.parse(jsonMatch)
   716→        } catch {

   718→        }
   719→      } else {
   720→        parsed = { raw: content }
   721→      }
   722→    }
   723→

   728→      { error: 'Error generando contenido', details: error.message },
