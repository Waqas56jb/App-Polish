    48→  const { brandProfile, setBrandProfile } = useAppStore()
    49→  const [activeTab, setActiveTab] = useState<Tab>('documento')
    50→  const [saved, setSaved] = useState(false)
    51→  const [activeMicField, setActiveMicField] = useState<string | null>(null)
    52→  const [isRecording, setIsRecording] = useState(false)
    53→
    54→  const [form, setForm] = useState<BrandProfile>(brandProfile || {
    55→    nombre: '',
    56→    salon: '',
    57→    ciudad: '',
    58→    instagram: '',
    59→    experiencia: '',
    60→    servicios: [],
    61→    serviciosPrioritarios: [],
    62→    objetivos: '',
    63→    clientaIdeal: '',
    64→    preguntasFrecuentes: '',
    65→    erroresFrecuentes: '',
    66→    nivelCamara: '',
    67→    facturacion: '',
    68→    documentText: '',
    69→    documentName: '',
    70→  })
    71→
    72→  // Document tab state
    73→  const [documentText, setDocumentText] = useState(form.documentText || '')
    74→  const [documentName, setDocumentName] = useState(form.documentName || '')
    75→  const [isExtracting, setIsExtracting] = useState(false)
    76→  const [extracted, setExtracted] = useState(false)
    77→  const fileInputRef = useRef<HTMLInputElement>(null)
    78→
    79→  // Guided tab state
    80→  const [guidedAnswers, setGuidedAnswers] = useState<Record<string, string>>({
    81→    nombre: form.nombre,
    82→    salon: form.salon,
    83→    ciudad: form.ciudad,
    84→    instagram: form.instagram,
    85→    experiencia: form.experiencia,

    88→    objetivos: form.objetivos,
    89→    clientaIdeal: form.clientaIdeal,
    90→    preguntasFrecuentes: form.preguntasFrecuentes,
    91→    erroresFrecuentes: form.erroresFrecuentes,
    92→    nivelCamara: form.nivelCamara,
    93→    facturacion: form.facturacion,
    94→  })
    95→
    96→  const updateField = (field: keyof BrandProfile, value: string | string[]) => {
    97→    setForm(prev => ({ ...prev, [field]: value }))
    98→    setSaved(false)
    99→  }
   100→
   101→  const toggleServicio = (servicio: string, field: 'servicios' | 'serviciosPrioritarios') => {
   102→    const current = form[field]
   103→    const maxItems = field === 'serviciosPrioritarios' ? 3 : 99
   104→    if (current.includes(servicio)) {
   105→      updateField(field, current.filter(s => s !== servicio))
   106→    } else if (current.length < maxItems) {
   107→      updateField(field, [...current, servicio])
   108→    }
   109→  }
   110→
   111→  const handleSave = () => {
   112→    setBrandProfile(form)
   113→    setSaved(true)
   114→    setTimeout(() => setSaved(false), 3000)
   115→  }
   116→
   117→  const handleMicClick = (fieldName: string, value: string, onChange: (v: string) => void) => {
   118→    // Use Web Speech API if available
   119→    const SpeechRecognition = (window as any).SpeechRecognition || (window as any).webkitSpeechRecognition
   120→
   121→    if (activeMicField === fieldName && isRecording) {
   122→      setIsRecording(false)
   123→      setActiveMicField(null)
   124→      return
   125→    }
   126→
   127→    if (!SpeechRecognition) {
   128→      // Fallback: simulate recording
   129→      setActiveMicField(fieldName)
   130→      setIsRecording(true)
   131→      setTimeout(() => {
   132→        setIsRecording(false)
   133→        setActiveMicField(null)
   134→      }, 2500)
   135→      return
   136→    }
   137→
   138→    const recognition = new SpeechRecognition()
   139→    recognition.lang = 'es-ES'
   140→    recognition.continuous = false
   141→    recognition.interimResults = false
   142→
   143→    recognition.onresult = (event: any) => {
   144→      const transcript = event.results[0][0].transcript
   145→      onChange(value ? value + ' ' + transcript : transcript)
   146→    }
   147→
   148→    recognition.onend = () => {
   149→      setIsRecording(false)
   150→      setActiveMicField(null)
   151→    }
   152→
   153→    recognition.onerror = () => {
   154→      setIsRecording(false)
   155→      setActiveMicField(null)
   156→    }
   157→
   158→    setActiveMicField(fieldName)
   159→    setIsRecording(true)
   160→    recognition.start()
   161→  }
   162→
   163→  // Document tab: handle file upload
   164→  const handleFileUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   165→    const file = e.target.files?.[0]
   166→    if (!file) return
   167→
   168→    setDocumentName(file.name)
   169→
   170→    if (file.type === 'text/plain' || file.type === 'text/markdown' || file.name.endsWith('.txt') || file.name.endsWith('.md')) {
   171→      const text = await file.text()
   172→      setDocumentText(text)
   173→    } else if (file.type === 'application/pdf' || file.name.endsWith('.pdf')) {
   174→      // For PDFs, show instructions to copy-paste
   175→      setDocumentText('')

   177→    } else {
   178→      // Try reading as text
   179→      try {
   180→        const text = await file.text()
   181→        setDocumentText(text)
   182→      } catch {
   183→        alert('Formato no soportado. Por favor copia y pega el texto directamente.')
   184→      }
   185→    }
   186→  }
   187→
   188→  // Document tab: extract structured data via AI
   189→  const handleExtract = async () => {
   190→    if (!documentText.trim()) return

   194→        method: 'POST',
   195→        headers: { 'Content-Type': 'application/json' },

   197→          type: 'extract-brand',
   198→          context: { documentText },
   199→        }),
   200→      })

   203→      if (data.result && !data.result.raw) {

   206→          ...form,
   207→          nombre: extracted.nombre || form.nombre,
   208→          salon: extracted.salon || form.salon,
   209→          ciudad: extracted.ciudad || form.ciudad,
   210→          instagram: extracted.instagram || form.instagram,
   211→          experiencia: extracted.experiencia || form.experiencia,
   212→          servicios: Array.isArray(extracted.servicios) ? extracted.servicios : (extracted.servicios ? [extracted.servicios] : form.servicios),
   213→          serviciosPrioritarios: Array.isArray(extracted.serviciosPrioritarios) ? extracted.serviciosPrioritarios : (extracted.serviciosPrioritarios ? [extracted.serviciosPrioritarios] : form.serviciosPrioritarios),
   214→          objetivos: extracted.objetivos || form.objetivos,
   215→          clientaIdeal: extracted.clientaIdeal || form.clientaIdeal,
   216→          preguntasFrecuentes: extracted.preguntasFrecuentes || form.preguntasFrecuentes,
   217→          erroresFrecuentes: extracted.erroresFrecuentes || form.erroresFrecuentes,
   218→          nivelCamara: extracted.nivelCamara || form.nivelCamara,
   219→          facturacion: extracted.facturacion || form.facturacion,
   220→          documentText,
   221→          documentName,
   222→        }
   223→        setForm(newForm)
   224→
   225→        // Update guided answers
   226→        setGuidedAnswers({
   227→          nombre: newForm.nombre,
   228→          salon: newForm.salon,
   229→          ciudad: newForm.ciudad,
   230→          instagram: newForm.instagram,
   231→          experiencia: newForm.experiencia,
   232→          servicios: newForm.servicios.join(', '),
   233→          serviciosPrioritarios: newForm.serviciosPrioritarios.join(', '),
   234→          objetivos: newForm.objetivos,
   235→          clientaIdeal: newForm.clientaIdeal,
   236→          preguntasFrecuentes: newForm.preguntasFrecuentes,
   237→          erroresFrecuentes: newForm.erroresFrecuentes,
   238→          nivelCamara: newForm.nivelCamara,
   239→          facturacion: newForm.facturacion,
   240→        })
   241→
   242→        setExtracted(true)
   243→        setTimeout(() => setExtracted(false), 4000)
   244→      } else {
   245→        alert('No se pudo extraer la información automáticamente. Por favor completa los campos manualmente en las otras pestañas.')
   246→      }
   247→    } catch (error) {
   248→      console.error('Extract error:', error)
   249→      // Save at least the document text so user can complete manually
   250→      setForm(prev => ({ ...prev, documentText, documentName }))
   251→      alert('Hubo un error al procesar con IA. Tu documento se ha guardado y puedes completar manualmente el formulario.')

   253→      setIsExtracting(false)
   254→    }
   255→  }
   256→
   257→  // Guided tab: save answers to form

   261→      nombre: guidedAnswers.nombre || form.nombre,
   262→      salon: guidedAnswers.salon || form.salon,
   263→      ciudad: guidedAnswers.ciudad || form.ciudad,
   264→      instagram: guidedAnswers.instagram || form.instagram,
   265→      experiencia: guidedAnswers.experiencia || form.experiencia,
   266→      servicios: guidedAnswers.servicios ? guidedAnswers.servicios.split(',').map(s => s.trim()).filter(Boolean) : form.servicios,

   271→      erroresFrecuentes: guidedAnswers.erroresFrecuentes || form.erroresFrecuentes,
   272→      nivelCamara: guidedAnswers.nivelCamara || form.nivelCamara,
   273→      facturacion: guidedAnswers.facturacion || form.facturacion,
   274→    }
   275→    setForm(newForm)
   276→    setBrandProfile(newForm)
   277→    setSaved(true)
   278→    setTimeout(() => setSaved(false), 3000)
   279→  }

   338→      {/* TAB 1: Document Upload / Paste */}
