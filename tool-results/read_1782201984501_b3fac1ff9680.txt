    50→  const { brandProfile, setBrandProfile } = useAppStore()

    52→  const [saved, setSaved] = useState(false)
    53→  const [activeMicField, setActiveMicField] = useState<string | null>(null)
    54→  const [isRecording, setIsRecording] = useState(false)
    55→
    56→  const [form, setForm] = useState<BrandProfile>(brandProfile || {
    57→    nombre: '',
    58→    salon: '',
    59→    ciudad: '',
    60→    instagram: '',
    61→    experiencia: '',
    62→    servicios: [],
    63→    serviciosPrioritarios: [],
    64→    objetivos: '',
    65→    clientaIdeal: '',
    66→    preguntasFrecuentes: '',
    67→    erroresFrecuentes: '',
    68→    nivelCamara: '',
    69→    facturacion: '',
    70→    documentText: '',

    73→
    74→  // Document tab state
    75→  const [documentText, setDocumentText] = useState(form.documentText || '')
    76→  const [documentName, setDocumentName] = useState(form.documentName || '')
    77→  const [isExtracting, setIsExtracting] = useState(false)
    78→  const [extracted, setExtracted] = useState(false)
    79→  const fileInputRef = useRef<HTMLInputElement>(null)
    80→
    81→  // Guided tab state
    82→  const [guidedAnswers, setGuidedAnswers] = useState<Record<string, string>>({
    83→    nombre: form.nombre,

    91→    clientaIdeal: form.clientaIdeal,
    92→    preguntasFrecuentes: form.preguntasFrecuentes,

    95→    facturacion: form.facturacion,

    97→
    98→  const updateField = (field: keyof BrandProfile, value: string | string[]) => {

   100→    setSaved(false)
   101→  }

   106→    if (current.includes(servicio)) {
   107→      updateField(field, current.filter(s => s !== servicio))

   109→      updateField(field, [...current, servicio])
   110→    }
   111→  }
   112→
   113→  const handleSave = () => {
   114→    setBrandProfile(form)
   115→    setSaved(true)
   116→    setTimeout(() => setSaved(false), 3000)
   117→  }
   118→
   119→  const handleMicClick = (fieldName: string, value: string, onChange: (v: string) => void) => {
   120→    // Use Web Speech API if available
   121→    const SpeechRecognition = (window as any).SpeechRecognition || (window as any).webkitSpeechRecognition
   122→
   123→    if (activeMicField === fieldName && isRecording) {
   124→      setIsRecording(false)
   125→      setActiveMicField(null)
   126→      return
   127→    }
   128→
   129→    if (!SpeechRecognition) {
   130→      // Fallback: simulate recording
   131→      setActiveMicField(fieldName)
   132→      setIsRecording(true)
   133→      setTimeout(() => {
   134→        setIsRecording(false)
   135→        setActiveMicField(null)
   136→      }, 2500)
   137→      return
   138→    }
   139→
   140→    const recognition = new SpeechRecognition()
   141→    recognition.lang = 'es-ES'
   142→    recognition.continuous = false
   143→    recognition.interimResults = false
   144→
   145→    recognition.onresult = (event: any) => {
   146→      const transcript = event.results[0][0].transcript
   147→      onChange(value ? value + ' ' + transcript : transcript)
   148→    }
   149→
   150→    recognition.onend = () => {
   151→      setIsRecording(false)
   152→      setActiveMicField(null)
   153→    }
   154→
   155→    recognition.onerror = () => {
   156→      setIsRecording(false)
   157→      setActiveMicField(null)
   158→    }
   159→
   160→    setActiveMicField(fieldName)
   161→    setIsRecording(true)
   162→    recognition.start()
   163→  }

   169→
   170→    setDocumentName(file.name)
   171→
   172→    if (file.type === 'text/plain' || file.type === 'text/markdown' || file.name.endsWith('.txt') || file.name.endsWith('.md')) {
   173→      const text = await file.text()

   175→    } else if (file.type === 'application/pdf' || file.name.endsWith('.pdf')) {

   181→      try {
   182→        const text = await file.text()

   186→      }

   210→          nombre: extracted.nombre || form.nombre,
   211→          salon: extracted.salon || form.salon,
   212→          ciudad: extracted.ciudad || form.ciudad,
   213→          instagram: extracted.instagram || form.instagram,
   214→          experiencia: extracted.experiencia || form.experiencia,
   215→          servicios: Array.isArray(extracted.servicios) ? extracted.servicios : (extracted.servicios ? [extracted.servicios] : form.servicios),
   216→          serviciosPrioritarios: Array.isArray(extracted.serviciosPrioritarios) ? extracted.serviciosPrioritarios : (extracted.serviciosPrioritarios ? [extracted.serviciosPrioritarios] : form.serviciosPrioritarios),
   217→          objetivos: extracted.objetivos || form.objetivos,
   218→          clientaIdeal: extracted.clientaIdeal || form.clientaIdeal,
   219→          preguntasFrecuentes: extracted.preguntasFrecuentes || form.preguntasFrecuentes,
   220→          erroresFrecuentes: extracted.erroresFrecuentes || form.erroresFrecuentes,
   221→          nivelCamara: extracted.nivelCamara || form.nivelCamara,
   222→          facturacion: extracted.facturacion || form.facturacion,
   223→          documentText,
   224→          documentName,
   225→        }

   227→
   228→        // Update guided answers
   229→        setGuidedAnswers({
   230→          nombre: newForm.nombre,
   231→          salon: newForm.salon,
   232→          ciudad: newForm.ciudad,
   233→          instagram: newForm.instagram,
   234→          experiencia: newForm.experiencia,
   235→          servicios: newForm.servicios.join(', '),
   236→          serviciosPrioritarios: newForm.serviciosPrioritarios.join(', '),
   237→          objetivos: newForm.objetivos,
   238→          clientaIdeal: newForm.clientaIdeal,
   239→          preguntasFrecuentes: newForm.preguntasFrecuentes,
   240→          erroresFrecuentes: newForm.erroresFrecuentes,
   241→          nivelCamara: newForm.nivelCamara,
   242→          facturacion: newForm.facturacion,
   243→        })
   244→
   245→        setExtracted(true)
   246→        setTimeout(() => setExtracted(false), 4000)
   247→      } else {
   248→        alert('No se pudo extraer la información automáticamente. Por favor completa los campos manualmente en las otras pestañas.')
   249→      }
   250→    } catch (error) {
   251→      console.error('Extract error:', error)
   252→      // Save at least the document text so user can complete manually
   253→      setForm(prev => ({ ...prev, documentText, documentName }))
   254→      alert('Hubo un error al procesar con IA. Tu documento se ha guardado y puedes completar manualmente el formulario.')
   255→    } finally {
   256→      setIsExtracting(false)
   257→    }
   258→  }
   259→
   260→  // Guided tab: save answers to form
   261→  const handleGuidedSave = () => {

   263→      ...form,
   264→      nombre: guidedAnswers.nombre || form.nombre,
   265→      salon: guidedAnswers.salon || form.salon,
   266→      ciudad: guidedAnswers.ciudad || form.ciudad,
   267→      instagram: guidedAnswers.instagram || form.instagram,
   268→      experiencia: guidedAnswers.experiencia || form.experiencia,
   269→      servicios: guidedAnswers.servicios ? guidedAnswers.servicios.split(',').map(s => s.trim()).filter(Boolean) : form.servicios,

   272→      clientaIdeal: guidedAnswers.clientaIdeal || form.clientaIdeal,
   273→      preguntasFrecuentes: guidedAnswers.preguntasFrecuentes || form.preguntasFrecuentes,
   274→      erroresFrecuentes: guidedAnswers.erroresFrecuentes || form.erroresFrecuentes,
   275→      nivelCamara: guidedAnswers.nivelCamara || form.nivelCamara,
   276→      facturacion: guidedAnswers.facturacion || form.facturacion,
   277→    }
   278→    setForm(newForm)
   279→    setBrandProfile(newForm)
   280→    setSaved(true)
   281→    setTimeout(() => setSaved(false), 3000)
   282→  }
   283→
   284→  const updateGuidedAnswer = (key: string, value: string) => {
   285→    setGuidedAnswers(prev => ({ ...prev, [key]: value }))
   286→    setSaved(false)
   287→  }
   288→

   290→
   291→  const tabs = [
   292→    { key: 'documento' as Tab, label: 'Subir / Pegar', icon: <FileUp className="w-4 h-4" /> },
   293→    { key: 'guiado' as Tab, label: 'Asistente Guiado', icon: <ListChecks className="w-4 h-4" /> },

   298→    <motion.div

   318→      {/* Info banner - not mandatory */}

   326→            </p>
   327→          </div>
   328→        </CardContent>
   329→      </Card>
   330→
   331→      {/* Tabs */}
   332→      <div className="flex gap-2 justify-center flex-wrap">
   333→        {tabs.map((tab) => (
   334→          <button
   335→            key={tab.key}
   336→            onClick={() => setActiveTab(tab.key)}
   337→            className={`brave-focus flex items-center gap-2 px-5 py-2.5 rounded-2xl font-medium transition-all duration-200 ${
   338→              activeTab === tab.key
   339→                ? 'bg-[#591427] text-white shadow-md brave-glow'

   342→          >
   343→            {tab.icon}
   344→            {tab.label}
   345→          </button>

   347→      </div>
   348→
   349→      {/* TAB 1: Document Upload / Paste */}
   350→      {activeTab === 'documento' && (
   351→        <motion.div

   362→                Sube o pega tu Documento de Marca

   390→              {/* Paste area */}

   406→                      onClick={() => { setDocumentText(''); setDocumentName(''); }}
   407→                      className="text-red-400 hover:text-red-600"
   408→                    >
   409→                      Limpiar
   410→                    </button>
   411→                  )}
   412→                </div>

   454→              </div>
   455→
   456→              {extracted && (

   458→                  <CardContent className="p-3 flex items-center gap-2">

   460→                    <p className="text-sm text-green-800">
   461→                      Información extraída con éxito. Revisa y completa lo que falte en las pestañas "Asistente Guiado" o "Formulario".
   462→                    </p>
   463→                  </CardContent>
   464→                </Card>

   469→          {/* Help card */}
