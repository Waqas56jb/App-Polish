    83→export function BancoGanchos() {
    84→  const {
    85→    brandProfile,
    86→    customHooks,
    87→    addCustomHook,
    88→    updateCustomHook,
    89→    removeCustomHook,
    90→    savedHooks,
    91→    saveHook,
    92→    updateSavedHook,
    93→    removeSavedHook,
    94→    addLibraryItem,
    95→    setIsLoading,
    96→  } = useAppStore()
    97→
    98→  const [vista, setVista] = useState<Vista>('banco')
    99→  const [categoriaActiva, setCategoriaActiva] = useState<string>('')
   100→  const [searchTerm, setSearchTerm] = useState('')
   101→  const [copiado, setCopiado] = useState<string | null>(null)
   102→
   103→  // Hook seleccionado para ver detalle/generar
   104→  const [hookSeleccionado, setHookSeleccionado] = useState<HookCard | null>(null)
   105→  const [modalOpen, setModalOpen] = useState(false)
   106→
   107→  // Generar contenido
   108→  const [generando, setGenerando] = useState(false)
   109→  const [resultadoGenerado, setResultadoGenerado] = useState<any>(null)
   110→
   111→  // Dialogo crear gancho
   112→  const [dialogoCrear, setDialogoCrear] = useState(false)
   113→
   114→  // Todos los ganchos
   115→  const todosGanchos = useMemo(() => {
   116→    return [...customHooks, ...HOOKS_SEED]
   117→  }, [customHooks])
   118→
   119→  // Ganchos filtrados
   120→  const ganchosFiltrados = useMemo(() => {
   121→    return todosGanchos.filter(g => {
   122→      if (categoriaActiva && g.categoria !== categoriaActiva) return false
   123→      if (searchTerm) {
   124→        const t = searchTerm.toLowerCase()
   125→        return g.titulo.toLowerCase().includes(t) || g.categoria.toLowerCase().includes(t) || g.servicio.toLowerCase().includes(t)
   126→      }
   127→      return true
   128→    })
   129→  }, [todosGanchos, categoriaActiva, searchTerm])
   130→
   131→  const yaGuardado = (hookId: string) => savedHooks.some(h => h.hookId === hookId)
   132→
   133→  const handleGuardar = (gancho: HookCard) => {
   134→    if (yaGuardado(gancho.id)) return
   135→    const nueva: SavedHook = {
   136→      id: generateId(),
   137→      hookId: gancho.id,
   138→      titulo: gancho.titulo,
   139→      tipoContenido: '',
   140→      servicio: gancho.servicio,
   141→      guionGenerado: '',
   142→      estado: 'idea',
   143→      fechaProgramada: null,
   144→      fechaGrabacion: null,
   145→      createdAt: new Date().toISOString(),
   146→    }
   147→    saveHook(nueva)
   148→    setCopiado('saved-' + gancho.id)
   149→    setTimeout(() => setCopiado(null), 1800)
   150→  }
   151→
   152→  const handleCopiarTitulo = (id: string, texto: string) => {
   153→    copyToClipboard(texto)
   154→    setCopiado(id)
   155→    setTimeout(() => setCopiado(null), 1500)
   156→  }
   157→
   158→  // Abrir hook para ver guion completo
   159→  const abrirHook = (gancho: HookCard) => {
   160→    setHookSeleccionado(gancho)
   161→    setResultadoGenerado(null)
   162→    setModalOpen(true)
   163→  }
   164→
   165→  // Generar contenido desde hook
   166→  const handleGenerar = async () => {
   167→    if (!hookSeleccionado) return
   168→    setGenerando(true)
   169→    setIsLoading(true, 'Generando contenido...')
   170→    try {
   171→      const res = await fetch('/api/ai', {
   172→        method: 'POST',
   173→        headers: { 'Content-Type': 'application/json' },
   174→        body: JSON.stringify({
   175→          type: 'generar-desde-gancho',
   176→          brandProfile,
   177→          context: {
   178→            gancho: hookSeleccionado.titulo,
   179→            tipoContenido: 'reel',
   180→            tono: 'cercano',
   181→            modo: 'camara',
   182→            servicio: hookSeleccionado.servicio,
   183→          },
   184→        }),
   185→      })
   186→      if (!res.ok) throw new Error('Network error')
   187→      const data = await res.json()
   188→      setResultadoGenerado(data.result)
   189→    } catch (e: any) {
   190→      console.error('Generate error:', e)
   191→    } finally {
   192→      setGenerando(false)
   193→      setIsLoading(false)
   194→    }
   195→  }
   196→
   197→  // Guardar generado en biblioteca
   198→  const handleGuardarGenerado = () => {
   199→    if (!hookSeleccionado || !resultadoGenerado) return
   200→    const titulo = resultadoGenerado.titulo || hookSeleccionado.titulo
   201→    const guionTexto = resultadoGenerado.guion || ''
   202→
   203→    // Guardar como SavedHook
   204→    const saved: SavedHook = {
   205→      id: generateId(),
   206→      hookId: hookSeleccionado.id,
   207→      titulo,
   208→      tipoContenido: 'reel',
   209→      servicio: hookSeleccionado.servicio,
   210→      guionGenerado: guionTexto + (resultadoGenerado.copy ? '\n\nCOPY:\n' + resultadoGenerado.copy : '') + (resultadoGenerado.hashtags ? '\n\n' + resultadoGenerado.hashtags : ''),
   211→      estado: 'pendiente',
   212→      fechaProgramada: null,
   213→      fechaGrabacion: null,
   214→      createdAt: new Date().toISOString(),
   215→    }
   216→    saveHook(saved)
   217→
   218→    // También en biblioteca general
   219→    addLibraryItem({
   220→      id: generateId(),
   221→      tipo: 'reel',
   222→      titulo,
   223→      objetivo: hookSeleccionado.objetivo,
   224→      servicio: hookSeleccionado.servicio,
   225→      guion: guionTexto,
   226→      copy: resultadoGenerado.copy || '',
   227→      hashtags: resultadoGenerado.hashtags || '',
   228→      textoPortada: resultadoGenerado.textoPortada || '',
   229→      formato: 'hablando a cámara',
   230→      estado: 'borrador',
   231→      fecha: '',
   232→      diaSemana: '',
   233→      slides: resultadoGenerado.slides || [],
   234→      storiesData: resultadoGenerado.stories || [],
   235→      descripcion: hookSeleccionado.titulo,
   236→      planId: '',
   237→      createdAt: new Date().toISOString(),
   238→    })
   239→
   240→    setCopiado('generado')
   241→    setTimeout(() => setCopiado(null), 1800)
   242→  }
   243→
   244→  return (
   245→    <div className="space-y-5">
   246→      {/* Cabecera compacta */}
   247→      <div className="flex items-center justify-between">
   248→        <div>
   249→          <h1 className="text-2xl font-bold text-foreground tracking-tight">Banco de Ganchos</h1>
   250→          <p className="text-sm text-muted-foreground mt-0.5">Ideas listas para usar. Elige, guarda y crea.</p>
   251→        </div>
   252→      </div>
   253→
   254→      {/* Selector de vista simple */}
   255→      <div className="flex gap-1.5 bg-card rounded-2xl p-1 shadow-sm border border-border w-fit">
   256→        {[
   257→          { key: 'banco' as Vista, label: 'Ganchos', icon: Lightbulb },
   258→          { key: 'mis-ideas' as Vista, label: 'Guardados', icon: Bookmark },
   259→          { key: 'calendario' as Vista, label: 'Calendario', icon: Calendar },
   260→        ].map(v => (
   261→          <button
   262→            key={v.key}
   263→            onClick={() => setVista(v.key)}
   264→            className={`flex items-center gap-1.5 px-4 py-2 rounded-xl text-sm font-medium transition-all ${
   265→              vista === v.key
   266→                ? 'brave-gradient text-white shadow-md'
   267→                : 'text-muted-foreground hover:text-foreground hover:bg-muted/50'
   268→            }`}
   269→          >
   270→            <v.icon className="w-4 h-4" />
   271→            {v.label}
   272→          </button>
   273→        ))}
   274→      </div>
   275→
   276→      {/* VISTA: BANCO */}
   277→      {vista === 'banco' && (
   278→        <div className="space-y-4">
   279→          {/* Búsqueda + categorías en una sola barra compacta */}
   280→          <div className="space-y-3">
   281→            <div className="relative">
   282→              <Search className="absolute left-3.5 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   283→              <Input
   284→                value={searchTerm}
   285→                onChange={(e) => setSearchTerm(e.target.value)}
   286→                placeholder="Buscar por título o tema..."
   287→                className="pl-10 bg-card border-border rounded-xl h-10"
   288→              />
   289→            </div>
   290→
   291→            {/* Chips de categorías horizontales - scrollable */}
   292→            <div className="flex gap-2 overflow-x-auto pb-1 -mx-1 px-1 no-scrollbar">
   293→              <button
   294→                onClick={() => setCategoriaActiva('')}
   295→                className={`shrink-0 px-3.5 py-1.5 rounded-full text-xs font-medium transition-all border ${
   296→                  !categoriaActiva
   297→                    ? 'bg-foreground text-background border-foreground'
   298→                    : 'bg-card text-muted-foreground border-border hover:bg-muted'
   299→                }`}
   300→              >
   301→                Todos
   302→              </button>
   303→              {CATEGORIAS_PRINCIPALES.map(cat => (
   304→                <button
   305→                  key={cat}
   306→                  onClick={() => setCategoriaActiva(cat === categoriaActiva ? '' : cat)}
   307→                  className={`shrink-0 px-3.5 py-1.5 rounded-full text-xs font-medium transition-all border ${
   308→                    cat === categoriaActiva
   309→                      ? 'bg-foreground text-background border-foreground'
   310→                      : `${CATEGORIA_COLORS[cat] || 'bg-card text-muted-foreground border-border'} hover:opacity-80`
   311→                  }`}
   312→                >
   313→                  {cat}
   314→                </button>
   315→              ))}
   316→            </div>
   317→          </div>
   318→
   319→          {/* Grid de tarjetas simplificado */}
   320→          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-3">
   321→            {ganchosFiltrados.map((gancho, idx) => {
   322→              const guardado = yaGuardado(gancho.id)
   323→              return (
   324→                <motion.div
   325→                  key={gancho.id}
   326→                  initial={{ opacity: 0, y: 12 }}
   327→                  animate={{ opacity: 1, y: 0 }}
   328→                  transition={{ duration: 0.25, delay: Math.min(idx * 0.03, 0.3) }}
   329→                >
   330→                  <Card className="bg-card border-border shadow-sm hover:shadow-md transition-all duration-200 group rounded-2xl overflow-hidden">
   331→                    <CardContent className="p-4 space-y-3">
   332→                      {/* Categoría badge */}
   333→                      <div className="flex items-center justify-between">
   334→                        <span className={`text-[10px] font-semibold px-2 py-0.5 rounded-full border ${CATEGORIA_COLORS[gancho.categoria] || 'bg-muted text-muted-foreground border-border'}`}>
   335→                          {gancho.categoria}
   336→                        </span>
   337→                        <span className="text-[10px] text-muted-foreground font-medium">
   338→                          {gancho.impacto} impacto
   339→                        </span>
   340→                      </div>
   341→
   342→                      {/* Gancho - texto principal */}
   343→                      <p className="text-sm font-semibold text-foreground leading-snug group-hover:text-primary transition-colors">
   344→                        &ldquo;{gancho.titulo}&rdquo;
   345→                      </p>
   346→
   347→                      {/* Explicación breve */}
   348→                      <p className="text-xs text-muted-foreground leading-relaxed line-clamp-2">
   349→                        {gancho.explicacion}
   350→                      </p>
   351→
   352→                      {/* Dos botones principales */}
   353→                      <div className="flex gap-2 pt-1">
   354→                        <Button
   355→                          size="sm"
   356→                          onClick={() => handleGuardar(gancho)}
   357→                          disabled={guardado}
   358→                          variant={guardado ? 'secondary' : 'default'}
   359→                          className={`flex-1 h-9 text-xs font-medium rounded-xl transition-all ${
   360→                            guardado
   361→                              ? 'bg-muted text-muted-foreground cursor-default'
   362→                              : 'bg-card hover:bg-muted border border-border text-foreground'
   363→                          }`}
   364→                        >
   365→                          {copiado === 'saved-' + gancho.id ? (
   366→                            <><Check className="w-3.5 h-3.5 mr-1.5 text-emerald-600" /> Guardado</>
   367→                          ) : guardado ? (
   368→                            <><Bookmark className="w-3.5 h-3.5 mr-1.5" /> En biblioteca</>
   369→                          ) : (
   370→                            <><Bookmark className="w-3.5 h-3.5 mr-1.5" /> Guardar</>
   371→                          )}
   372→                        </Button>
   373→
   374→                        <Button
   375→                          size="sm"
   376→                          onClick={() => abrirHook(gancho)}
   377→                          className="flex-1 h-9 text-xs font-medium rounded-xl brave-gradient hover:opacity-90 text-white"
   378→                        >
   379→                          <FileText className="w-3.5 h-3.5 mr-1.5" />
   380→                          Ver guion
   381→                          <ChevronRight className="w-3 h-3 ml-0.5" />
   382→                        </Button>
   383→                      </div>
   384→                    </CardContent>
   385→                  </Card>
   386→                </motion.div>
   387→              )
   388→            })}
   389→          </div>
   390→
   391→          {ganchosFiltrados.length === 0 && (
   392→            <div className="text-center py-16">
   393→              <BravyBot size={56} expression="thinking" animate />
   394→              <p className="text-foreground font-medium mt-3 text-sm">No se encontraron ganchos</p>
   395→              <p className="text-xs text-muted-foreground mt-1">Prueba con otra categoría o búsqueda</p>
   396→            </div>
   397→          )}
   398→        </div>
   399→      )}
   400→
   401→      {/* VISTA: MIS IDEAS GUARDADAS */}
   402→      {vista === 'mis-ideas' && (
   403→        <MisIdeasGuardadas
   404→          savedHooks={savedHooks}
   405→          onUpdate={updateSavedHook}
   406→          onRemove={removeSavedHook}
   407→        />
   408→      )}
   409→
   410→      {/* VISTA: CALENDARIO */}
   411→      {vista === 'calendario' && (
   412→        <CalendarioSimple
   413→          savedHooks={savedHooks}
   414→          onUpdate={updateSavedHook}
   415→        />
   416→      )}
   417→
   418→      {/* MODAL: Ver guion completo del hook */}
   419→      <AnimatePresence>
   420→        {modalOpen && hookSeleccionado && (
   421→          <ModalGuion
   422→            gancho={hookSeleccionado}
   423→            resultadoGenerado={resultadoGenerado}
   424→            generando={generando}
   425→            copiado={copiado}
   426→            guardado={yaGuardado(hookSeleccionado.id)}
   427→            onClose={() => {
   428→              setModalOpen(false)
   429→              setHookSeleccionado(null)
   430→              setResultadoGenerado(null)
   431→            }}
   432→            onGuardar={() => handleGuardar(hookSeleccionado!)}
   433→            onGenerar={handleGenerar}
   434→            onGuardarGenerado={handleGuardarGenerado}
   435→            onCopiar={(field, text) => {
   436→              copyToClipboard(text)
   437→              setCopiado(field)
   438→              setTimeout(() => setCopiado(null), 1500)
   439→            }}
   440→          />
   441→        )}
   442→      </AnimatePresence>
   443→    </div>
   444→  )
   445→}

   450→function ModalGuion({
   451→  gancho,
   452→  resultadoGenerado,
   453→  generando,
   454→  copiado,
   455→  guardado,
   456→  onClose,
   457→  onGuardar,
   458→  onGenerar,
   459→  onGuardarGenerado,
   460→  onCopiar,
   461→}: {
   462→  gancho: HookCard
   463→  resultadoGenerado: any
   464→  generando: boolean
   465→  copiado: string | null
   466→  guardado: boolean
   467→  onClose: () => void
   468→  onGuardar: () => void
   469→  onGenerar: () => void
   470→  onGuardarGenerado: () => void
   471→  onCopiar: (field: string, text: string) => void
   472→}) {
   473→  const [showDetalle, setShowDetalle] = useState(false)
   474→
   475→  return (
   476→    <motion.div
   477→      initial={{ opacity: 0 }}
   478→      animate={{ opacity: 1 }}
   479→      exit={{ opacity: 0 }}
   480→      className="fixed inset-0 z-50 flex items-center justify-center bg-black/30 backdrop-blur-sm p-4"
   481→      onClick={onClose}
   482→    >
   483→      <motion.div
   484→        initial={{ scale: 0.95, opacity: 0, y: 20 }}
   485→        animate={{ scale: 1, opacity: 1, y: 0 }}
   486→        exit={{ scale: 0.95, opacity: 0, y: 20 }}
   487→        transition={{ type: 'spring', stiffness: 300, damping: 28 }}
   488→        className="bg-card w-full max-w-2xl rounded-3xl shadow-2xl max-h-[90vh] overflow-hidden flex flex-col border border-border"
   489→        onClick={(e) => e.stopPropagation()}
   490→      >
   491→        {/* Header */}
   492→        <div className="brave-gradient p-5 text-white relative">
   493→          <button
   494→            onClick={onClose}
   495→            className="absolute top-4 right-4 w-8 h-8 rounded-full bg-white/15 hover:bg-white/25 flex items-center justify-center transition-colors"
   496→          >
   497→            <X className="w-4 h-4" />
   498→          </button>
   499→          <div className="flex items-center gap-2 mb-2">
   500→            <span className={`text-[10px] font-semibold px-2 py-0.5 rounded-full bg-white/20`}>
   501→              {gancho.categoria}
   502→            </span>
   503→            <span className="text-[10px] px-2 py-0.5 rounded-full bg-white/20">
   504→              {gancho.tipo} · {gancho.objetivo}
   505→            </span>
   506→          </div>
   507→          <h2 className="text-xl font-bold leading-snug pr-8">
   508→            &ldquo;{gancho.titulo}&rdquo;
   509→          </h2>
   510→        </div>
   511→
   512→        {/* Body */}
   513→        <div className="p-5 overflow-y-auto flex-1 space-y-4">
   514→          {/* Info rápida */}
   515→          <div className="grid grid-cols-2 gap-2.5">
   516→            <div className="bg-muted/50 rounded-xl p-3">
   517→              <p className="text-[10px] uppercase tracking-wider text-muted-foreground mb-0.5">Servicio</p>
   518→              <p className="text-sm font-medium text-foreground">{gancho.servicio}</p>
   519→            </div>
   520→            <div className="bg-muted/50 rounded-xl p-3">

   896→// SUBCOMPONENTE: Calendario simple
   897→// ============================================================
   898→function CalendarioSimple({
   899→  savedHooks,
   900→  onUpdate,
   901→}: {
   902→  savedHooks: SavedHook[]
   903→  onUpdate: (id: string, updates: Partial<SavedHook>) => void
   904→}) {
   905→  const [mes, setMes] = useState(() => {
   906→    const d = new Date()
   907→    return new Date(d.getFullYear(), d.getMonth(), 1)
   908→  })
   909→
   910→  const programados = savedHooks.filter(h => h.fechaProgramada)
   911→  const porFecha = new Map<string, SavedHook[]>()
   912→  programados.forEach(h => {
   913→    if (!h.fechaProgramada) return
   914→    const arr = porFecha.get(h.fechaProgramada) || []
   915→    arr.push(h)
   916→    porFecha.set(h.fechaProgramada, arr)
   917→  })
   918→
   919→  const inicioMes = new Date(mes.getFullYear(), mes.getMonth(), 1)
   920→  const finMes = new Date(mes.getFullYear(), mes.getMonth() + 1, 0)
   921→  const inicioGrid = new Date(inicioMes)
   922→  inicioGrid.setDate(inicioGrid.getDate() - inicioGrid.getDay())
   923→  const dias: Date[] = []
   924→  const cursor = new Date(inicioGrid)
   925→  while (cursor <= finMes || dias.length % 7 !== 0) {
   926→    dias.push(new Date(cursor))
   927→    cursor.setDate(cursor.getDate() + 1)
   928→    if (dias.length > 42) break
   929→  }
   930→
   931→  const monthName = mes.toLocaleDateString('es-ES', { month: 'long', year: 'numeric' })
   932→
   933→  if (programados.length === 0) {
   934→    return (
   935→      <div className="text-center py-16">
   936→        <BravyBot size={56} expression="wave" animate />
   937→        <p className="text-foreground font-medium mt-3 text-sm">Sin contenidos programados</p>
   938→        <p className="text-xs text-muted-foreground mt-1">Asigna fechas a tus ideas guardadas para verlas aquí</p>
   939→      </div>
   940→    )
   941→  }
   942→
   943→  return (
   944→    <div className="space-y-4">
   945→      <div className="flex items-center gap-2">
   946→        <Button
   947→          size="sm"
   948→          variant="outline"
   949→          onClick={() => setMes(new Date(mes.getFullYear(), mes.getMonth() - 1, 1))}
   950→          className="border-border text-foreground h-8 rounded-lg"
   951→        >
   952→          ←
   953→        </Button>
   954→        <span className="text-sm font-semibold capitalize text-foreground min-w-[140px] text-center">{monthName}</span>
   955→        <Button
   956→          size="sm"
   957→          variant="outline"
   958→          onClick={() => setMes(new Date(mes.getFullYear(), mes.getMonth() + 1, 1))}
   959→          className="border-border text-foreground h-8 rounded-lg"
   960→        >
   961→          →
   962→        </Button>
   963→      </div>
   964→
   965→      <Card className="bg-card border-border shadow-sm rounded-2xl">
   966→        <CardContent className="p-4">
   967→          <div className="grid grid-cols-7 gap-1 mb-2">
   968→            {['D', 'L', 'M', 'X', 'J', 'V', 'S'].map(d => (
   969→              <div key={d} className="text-center text-[10px] font-semibold uppercase text-muted-foreground py-1">{d}</div>
   970→            ))}
   971→          </div>
   972→          <div className="grid grid-cols-7 gap-1">
   973→            {dias.map((dia, i) => {
   974→              const fechaStr = dia.toISOString().split('T')[0]
   975→              const enMes = dia.getMonth() === mes.getMonth()
   976→              const hooksHoy = porFecha.get(fechaStr) || []
   977→              const esHoy = new Date().toDateString() === dia.toDateString()
   978→              return (
   979→                <div
   980→                  key={i}
   981→                  className={`min-h-[60px] p-1.5 rounded-xl border ${
   982→                    enMes
   983→                      ? 'bg-card border-border'
   984→                      : 'bg-muted/30 border-transparent text-muted-foreground/40'
   985→                  } ${esHoy ? 'ring-2 ring-primary/20 bg-primary/5' : ''}`}
   986→                >
   987→                  <p className="text-[10px] font-medium text-foreground mb-0.5">{dia.getDate()}</p>
   988→                  <div className="space-y-0.5">
   989→                    {hooksHoy.slice(0, 2).map(h => (
   990→                      <div
   991→                        key={h.id}
   992→                        className={`text-[8px] px-1 py-0.5 rounded-md truncate ${
   993→                          h.estado === 'publicado'
   994→                            ? 'bg-emerald-100 text-emerald-800'
   995→                            : h.estado === 'grabado'
   996→                            ? 'bg-violet-100 text-violet-800'
   997→                            : 'bg-amber-100 text-amber-800'
   998→                        }`}
   999→                        title={h.titulo}
  1000→                      >
  1001→                        {h.titulo}
  1002→                      </div>
  1003→                    ))}
  1004→                    {hooksHoy.length > 2 && (
  1005→                      <p className="text-[8px] text-muted-foreground">+{hooksHoy.length - 2}</p>
  1006→                    )}
  1007→                  </div>
