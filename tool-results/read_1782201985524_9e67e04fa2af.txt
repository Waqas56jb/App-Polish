   111→// Extracts the CTA keyword from the last story (usually Story 3 contains "escribe KEYWORD")
   112→function extractKeywordFromStories(stories: BraveStory[]): string {
   113→  if (!stories || stories.length === 0) return ''
   114→  const lastStory = stories[stories.length - 1]
   115→  const match = (lastStory.texto || '').match(/escribe\s+([A-ZÁÉÍÓÚÑ][A-ZÁÉÍÓÚÑ\s]{1,20})/i)
   116→  if (match) return match[1].trim().toUpperCase()
   117→  return ''
   118→}
   119→
   120→function getStoryTipoVista(story: BraveStory, totalStories: number) {
   121→  if (story.numero === 1) return VISTAS_TIPO_STORY[0]
   122→  if (story.numero === totalStories) return VISTAS_TIPO_STORY[2]
   123→  return VISTAS_TIPO_STORY[1]
   124→}

   193→function SecuenciaStories() {
   194→  const { brandProfile, addLibraryItems, setIsLoading, isLoading } = useAppStore()
   195→
   196→  // Input state
   197→  const [servicio, setServicio] = useState('')
   198→  const [trabajoRealizado, setTrabajoRealizado] = useState('')
   199→  const [descripcionExtra, setDescripcionExtra] = useState('')
   200→  const [modo, setModo] = useState<'texto' | 'camara'>('texto')
   201→  const [numStories, setNumStories] = useState<number>(3)
   202→
   203→  // Audio recording state
   204→  const [isRecording, setIsRecording] = useState(false)
   205→  const [audioBlobUrl, setAudioBlobUrl] = useState<string | null>(null)
   206→  const [isTranscribing, setIsTranscribing] = useState(false)
   207→  const mediaRecorderRef = useRef<MediaRecorder | null>(null)
   208→  const audioChunksRef = useRef<Blob[]>([])
   209→
   210→  // Output state
   211→  const [resultado, setResultado] = useState<BraveStoryResult | null>(null)
   212→  const [error, setError] = useState<string | null>(null)
   213→  const [copiedField, setCopiedField] = useState<string | null>(null)
   214→  const [savedToLibrary, setSavedToLibrary] = useState(false)
   215→
   216→  const copyField = (field: string, text: string) => {
   217→    if (!text) return
   218→    navigator.clipboard.writeText(text)
   219→    setCopiedField(field)
   220→    setTimeout(() => setCopiedField(null), 2000)
   221→  }
   222→
   223→  // Audio recording handlers
   224→  const startRecording = async () => {
   225→    try {
   226→      const stream = await navigator.mediaDevices.getUserMedia({ audio: true })
   227→      const mediaRecorder = new MediaRecorder(stream, { mimeType: 'audio/webm' })
   228→      mediaRecorderRef.current = mediaRecorder
   229→      audioChunksRef.current = []
   230→
   231→      mediaRecorder.ondataavailable = (event) => {
   232→        if (event.data.size > 0) audioChunksRef.current.push(event.data)
   233→      }
   234→      mediaRecorder.onstop = () => {
   235→        const blob = new Blob(audioChunksRef.current, { type: 'audio/webm' })
   236→        const url = URL.createObjectURL(blob)
   237→        setAudioBlobUrl(url)
   238→        stream.getTracks().forEach(track => track.stop())
   239→      }
   240→      mediaRecorder.start()
   241→      setIsRecording(true)
   242→    } catch (e: any) {
   243→      setError('No se pudo acceder al micrófono: ' + e.message)
   244→    }
   245→  }
   246→
   247→  const stopRecording = () => {
   248→    if (mediaRecorderRef.current && isRecording) {
   249→      mediaRecorderRef.current.stop()
   250→      setIsRecording(false)
   251→    }
   252→  }
   253→
   254→  const transcribeAudio = async () => {
   255→    if (!audioBlobUrl) return
   256→    setIsTranscribing(true)
   257→    setError(null)
   258→    try {
   259→      const response = await fetch(audioBlobUrl)
   260→      const blob = await response.blob()
   261→      const reader = new FileReader()
   262→      const base64 = await new Promise<string>((resolve) => {
   263→        reader.onloadend = () => {
   264→          const result = reader.result as string
   265→          // Strip data URL prefix
   266→          resolve(result.split(',')[1])
   267→        }
   268→        reader.readAsDataURL(blob)
   269→      })
   270→
   271→      const res = await fetch('/api/asr', {
   272→        method: 'POST',
   273→        headers: { 'Content-Type': 'application/json' },
   274→        body: JSON.stringify({ audioBase64: base64 }),
   275→      })
   276→      const data = await res.json()
   277→      if (data.error) {
   278→        setError('Error transcribiendo audio: ' + data.error)
   279→      } else if (data.text) {
   280→        setTrabajoRealizado(prev => prev ? prev + ' ' + data.text : data.text)
   281→      } else {
   282→        setError('No se pudo transcribir el audio.')
   283→      }
   284→    } catch (e: any) {
   285→      setError('Error transcribiendo: ' + e.message)
   286→    } finally {
   287→      setIsTranscribing(false)
   288→    }
   289→  }
   290→
   291→  const clearAudio = () => {
   292→    if (audioBlobUrl) URL.revokeObjectURL(audioBlobUrl)
   293→    setAudioBlobUrl(null)
   294→    audioChunksRef.current = []
   295→  }
   296→
   297→  // Cleanup audio on unmount
   298→  useEffect(() => {
   299→    return () => {
   300→      if (audioBlobUrl) URL.revokeObjectURL(audioBlobUrl)
   301→    }
   302→  }, [audioBlobUrl])
   303→
   304→  const handleGenerate = async () => {
   305→    setError(null)
   306→    if (!servicio && !trabajoRealizado) {
   307→      setError('Cuéntanos qué trabajo realizaste hoy para crear tus Stories BRÄVE.')
   308→      return
   309→    }
   310→
   311→    setIsLoading(true, `Creando tus ${numStories} Stories BRÄVE...`)
   312→    setSavedToLibrary(false)
   313→    try {
   314→      const res = await fetch('/api/ai', {
   315→        method: 'POST',
   316→        headers: { 'Content-Type': 'application/json' },
   317→        body: JSON.stringify({
   318→          type: 'stories-brave',
   319→          brandProfile,
   320→          context: {
   321→            trabajoRealizado: trabajoRealizado || servicio,
   322→            servicio,
   323→            modo,
   324→            descripcionExtra,
   325→            numStories,
   326→          },
   327→        }),
   328→      })
   329→      const data = await res.json()
   330→      if (data.error) {
   331→        setError(data.error)
   332→      } else {
   333→        const r = data.result
   334→        if (Array.isArray(r) && r.length > 0 && r[0].numero) {
   335→          setResultado({
   336→            trabajo: servicio || trabajoRealizado || 'trabajo de hoy',
   337→            problemaCliente: '',
   338→            palabraClave: extractKeywordFromStories(r),
   339→            stories: r,
   340→            hashtags: '',
   341→          })
   342→        } else if (r && r.stories && Array.isArray(r.stories)) {
   343→          setResultado(r as BraveStoryResult)
   344→        } else if (r?.raw) {
   345→          setError('No se pudo procesar la respuesta. Intenta de nuevo.')
   346→        } else {
   347→          setError('Formato inesperado. Intenta de nuevo.')
   348→        }
   349→      }
   350→    } catch (e: any) {
   351→      setError(e.message || 'Error generando stories')
   352→    } finally {
   353→      setIsLoading(false)
   354→    }
   355→  }
   356→
   357→  const handleSaveToLibrary = () => {
   358→    if (!resultado) return
   359→    const item: ContentItem = {
   360→      id: generateId(),
   361→      tipo: 'story',
   362→      titulo: `Stories BRÄVE — ${servicio || resultado.trabajo || 'trabajo de hoy'}`,
   363→      objetivo: 'reservas',
   364→      servicio: servicio || '',
   365→      guion: resultado.stories.map(s => {
   366→        let out = `STORY ${s.numero} — ${s.tipo}\n${s.texto}`
   367→        if (s.encuesta) {
   368→          out += `\n\n📊 Encuesta: ${s.encuesta.pregunta}\n   Opciones: ${s.encuesta.respuestas.join(' / ')}`
   369→        }
   370→        out += `\n\nVisual: ${s.ideaVisual}\nSticker: ${s.sticker}`
   371→        return out
   372→      }).join('\n\n---\n\n'),
   373→      copy: '',
   374→      hashtags: resultado.hashtags || '',
   375→      textoPortada: resultado.problemaCliente || '',
   376→      formato: modo === 'camara' ? 'hablando a cámara' : 'texto',
   377→      estado: 'aprobado',
   378→      fecha: '',
   379→      diaSemana: '',
   380→      slides: [],
   381→      storiesData: resultado.stories.map(s => ({
   382→        numero: s.numero,
   383→        texto: s.texto,
   384→        sticker: s.sticker,
   385→        ideaVisual: s.ideaVisual,
   386→      })),
   387→      descripcion: resultado.problemaCliente || '',
   388→      planId: '',
   389→      createdAt: new Date().toISOString(),
   390→    }
   391→    addLibraryItems([item])
   392→    setSavedToLibrary(true)
   393→    setTimeout(() => setSavedToLibrary(false), 2500)
   394→  }
   395→
   396→  const handleCopyAll = () => {
   397→    if (!resultado) return
   398→    const lines: string[] = [
   399→      `STORIES BRÄVE — ${servicio || resultado.trabajo || 'trabajo de hoy'}`,
   400→      ``,
   401→    ]
   402→    resultado.stories.forEach(s => {
   403→      lines.push(`━━━━━━━━━━━━━━━━━━━━`)
   404→      lines.push(`STORY ${s.numero} — ${s.tipo}`)
   405→      lines.push(`━━━━━━━━━━━━━━━━━━━━`)
   406→      lines.push(s.texto)
   407→      if (s.encuesta) {
   408→        lines.push(``)
   409→        lines.push(`Encuesta: ${s.encuesta.pregunta}`)
   410→        lines.push(`Opciones: ${s.encuesta.respuestas.map((r, i) => `${i + 1}. ${r}`).join('  |  ')}`)
   411→      }
   412→      lines.push(``)
   413→      lines.push(`Visual: ${s.ideaVisual}`)
   414→      lines.push(`Sticker: ${s.sticker}`)
   415→      lines.push(``)
   416→    })
   417→    if (resultado.hashtags) lines.push(resultado.hashtags)
   418→    copyField('all', lines.join('\n'))
   419→  }
   420→
   421→  return (
   422→    <div className="space-y-6">
   423→      {/* Methodology callout */}
   424→      <Card className="border-[#E8DDD5] bg-[#FFFBF0]">
   425→        <CardContent className="p-5">
   426→          <div className="flex items-start gap-3">
   427→            <div className="w-9 h-9 rounded-full bg-[#FFF1B5] flex items-center justify-center shrink-0">
   428→              <Lightbulb className="w-4 h-4 text-white" />
   429→            </div>
   430→            <div className="flex-1">
   431→              <p className="text-sm font-bold text-[#591427] mb-1">Metodología BRÄVE</p>
   432→              <p className="text-sm text-[#2A1520]/80 mb-3">
   433→                Las Stories BRÄVE no se construyen alrededor del servicio. Se construyen alrededor de la clienta.
   434→                Las clientas no reservan porque vean un balayage bonito. Reservan porque se identifican con un problema
   435→                y entienden que existe una solución para ellas.
   436→              </p>
   437→              <div className="grid grid-cols-3 gap-2 text-xs">
   438→                {VISTAS_TIPO_STORY.map(v => (
   439→                  <div key={v.n} className="bg-white rounded-lg p-2 border border-[#E8DDD5]">
   440→                    <div className="flex items-center gap-1.5 mb-1">
   441→                      <span className={`w-5 h-5 rounded-full ${v.color} text-white text-[10px] font-bold flex items-center justify-center`}>
   442→                        {v.n}
   443→                      </span>
   444→                      <span className="font-semibold text-[#2A1520] truncate">{v.titulo}</span>
   445→                    </div>
   446→                    <p className="text-[#2A1520]/70 text-[11px] leading-tight">{v.descripcion}</p>
   447→                  </div>
   448→                ))}
   449→              </div>
   450→            </div>
   451→          </div>
   452→        </CardContent>
   453→      </Card>
   454→
   455→      {/* Input form */}
   456→      <Card className="border-[#E8DDD5]">
   457→        <CardHeader>
   458→          <CardTitle className="text-[#591427]">Cuéntanos qué hiciste hoy</CardTitle>
   459→          <CardDescription>
   460→            Solo necesitas decir lo que hiciste en el salón. BRÄVE se encarga del resto.
   461→          </CardDescription>
   462→        </CardHeader>
   463→        <CardContent className="space-y-5">
   464→          {/* Service selector */}
   465→          <div className="space-y-2">
   466→            <label className="text-sm font-semibold text-[#2A1520]">¿Qué trabajo realizaste hoy?</label>
   467→            <div className="flex flex-wrap gap-2">
   468→              {SERVICIOS_PREDEFINIDOS.map(s => (
   469→                <button
   470→                  key={s}
   471→                  onClick={() => {
   472→                    setServicio(s)
   473→                    if (!trabajoRealizado) setTrabajoRealizado(s)

   475→                  className={`px-3 py-1.5 rounded-full text-xs font-medium transition-all ${
   476→                    servicio === s
   477→                      ? 'bg-[#591427] text-white shadow-md'
   478→                      : 'bg-[#FFFBF0] text-[#2A1520] border border-[#E8DDD5] hover:border-[#C1DBE8]'
   479→                  }`}
   480→                >
   481→                  {s}
   482→                </button>
   483→              ))}
   484→            </div>
   485→          </div>
   486→
   487→          {/* Free text describing the work */}
   488→          <div className="space-y-2">
   489→            <label className="text-sm font-semibold text-[#2A1520]">Cuéntalo con tus palabras (opcional)</label>
   490→            <Textarea
   491→              value={trabajoRealizado}
   492→              onChange={e => setTrabajoRealizado(e.target.value)}
   493→              placeholder="Ej: Hoy vino una clienta que llevaba meses sin tocar su color. Le hicimos un balayage suave para devolverle luz sin pasar por rubia."
   494→              className="min-h-[80px] resize-none border-[#E8DDD5] focus:border-[#C1DBE8]"
   495→            />
   496→          </div>
   497→
   498→          {/* Audio input */}
   499→          <div className="space-y-2">
   500→            <label className="text-sm font-semibold text-[#2A1520]">O cuéntalo en voz alta (opcional)</label>
   501→            <div className="bg-[#FFFBF0] border border-[#E8DDD5] rounded-xl p-4">
   502→              {!isRecording && !audioBlobUrl && (
   503→                <Button
   504→                  onClick={startRecording}
   505→                  variant="outline"
   506→                  className="border-[#C1DBE8] text-[#C1DBE8] hover:bg-[#F5F0EB]"
   507→                >
   508→                  <Mic className="w-4 h-4 mr-2" />
   509→                  Grabar audio
   510→                </Button>
   511→              )}
   512→              {isRecording && (
   513→                <div className="flex items-center gap-3">
   514→                  <span className="flex items-center gap-2 text-sm text-red-600">
   515→                    <span className="w-3 h-3 rounded-full bg-red-500 animate-pulse" />
   516→                    Grabando...
   517→                  </span>
   518→                  <Button
   519→                    onClick={stopRecording}
   520→                    variant="outline"
   521→                    className="border-red-500 text-red-600 hover:bg-red-50"
   522→                  >
   523→                    <Square className="w-4 h-4 mr-2" />
   524→                    Detener
   525→                  </Button>
   526→                </div>
   527→              )}
   528→              {audioBlobUrl && !isRecording && (
   529→                <div className="space-y-3">
   530→                  <audio src={audioBlobUrl} controls className="w-full" />
   531→                  <div className="flex flex-wrap gap-2">
   532→                    <Button

   534→                      disabled={isTranscribing}

   537→                    >

   881→
   882→function CajaDePreguntas() {
   883→  const { brandProfile, addLibraryItems, setIsLoading, isLoading } = useAppStore()
   884→
   885→  // Input state
   886→  const [serviciosSeleccionados, setServiciosSeleccionados] = useState<string[]>([])
   887→  const [servicioCustom, setServicioCustom] = useState('')
   888→  const [numPreguntas, setNumPreguntas] = useState<number>(5)
   889→  const [modoRespuesta, setModoRespuesta] = useState<'texto' | 'camara'>('texto')
   890→
   891→  // Output state
   892→  const [resultado, setResultado] = useState<CajaPreguntasResult | null>(null)
   893→  const [error, setError] = useState<string | null>(null)
   894→  const [copiedField, setCopiedField] = useState<string | null>(null)
   895→  const [savedToLibrary, setSavedToLibrary] = useState(false)
   896→
   897→  const toggleServicio = (s: string) => {
   898→    setServiciosSeleccionados(prev =>
   899→      prev.includes(s) ? prev.filter(x => x !== s) : [...prev, s]
   900→    )
   901→  }
   902→
   903→  const addCustomServicio = () => {
   904→    const s = servicioCustom.trim()
   905→    if (s && !serviciosSeleccionados.includes(s)) {
   906→      setServiciosSeleccionados(prev => [...prev, s])
   907→      setServicioCustom('')
   908→    }
   909→  }
   910→
   911→  const copyField = (field: string, text: string) => {
   912→    if (!text) return
   913→    navigator.clipboard.writeText(text)
   914→    setCopiedField(field)
   915→    setTimeout(() => setCopiedField(null), 2000)
   916→  }
   917→
   918→  const handleGenerate = async () => {
   919→    setError(null)
   920→    if (serviciosSeleccionados.length === 0) {
   921→      setError('Selecciona al menos un servicio sobre el que quieras responder preguntas.')
   922→      return
   923→    }
   924→
   925→    setIsLoading(true, `Generando ${numPreguntas} preguntas para tu caja...`)
   926→    setSavedToLibrary(false)
   927→    try {
   928→      const res = await fetch('/api/ai', {
   929→        method: 'POST',
   930→        headers: { 'Content-Type': 'application/json' },
   931→        body: JSON.stringify({
   932→          type: 'preguntas-caja',
   933→          brandProfile,
   934→          context: {
   935→            servicios: serviciosSeleccionados,
   936→            numPreguntas,
   937→            modo: modoRespuesta,
   938→          },
   939→        }),
   940→      })
   941→      const data = await res.json()
   942→      if (data.error) {
   943→        setError(data.error)
   944→      } else {
   945→        const r = data.result
   946→        if (r && r.preguntas && Array.isArray(r.preguntas)) {
   947→          setResultado(r as CajaPreguntasResult)
   948→        } else if (Array.isArray(r) && r.length > 0 && r[0].pregunta) {
   949→          // Model returned just the preguntas array
   950→          setResultado({
   951→            servicios: serviciosSeleccionados,
   952→            preguntas: r,
   953→          })
   954→        } else if (r?.raw) {
   955→          setError('No se pudo procesar la respuesta. Intenta de nuevo.')
   956→        } else {
   957→          setError('Formato inesperado. Intenta de nuevo.')
   958→        }
   959→      }
   960→    } catch (e: any) {
   961→      setError(e.message || 'Error generando preguntas')
   962→    } finally {
   963→      setIsLoading(false)
   964→    }
   965→  }
   966→
   967→  const handleCopyAll = () => {
   968→    if (!resultado) return
   969→    const lines: string[] = [
   970→      `CAJA DE PREGUNTAS — Servicios: ${resultado.servicios.join(', ')}`,
   971→      ``,
   972→    ]
   973→    resultado.preguntas.forEach(p => {
   974→      lines.push(`━━━━━━━━━━━━━━━━━━━━`)
   975→      lines.push(`PREGUNTA ${p.id}`)
   976→      lines.push(`━━━━━━━━━━━━━━━━━━━━`)
   977→      lines.push(`Servicio: ${p.servicio}`)
   978→      lines.push(``)
   979→      lines.push(`P: ${p.pregunta}`)
   980→      lines.push(``)
   981→      lines.push(`R: ${p.respuesta}`)
   982→      lines.push(``)
   983→    })
   984→    copyField('all', lines.join('\n'))
   985→  }
   986→
   987→  const handleSaveToLibrary = () => {
   988→    if (!resultado) return
   989→    const items: ContentItem[] = resultado.preguntas.map(p => ({
   990→      id: generateId(),
   991→      tipo: 'story' as const,
   992→      titulo: `Pregunta caja — ${p.servicio}`,
   993→      objetivo: 'autoridad',
   994→      servicio: p.servicio,
   995→      guion: `PREGUNTA:\n${p.pregunta}\n\nRESPUESTA:\n${p.respuesta}`,
   996→      copy: p.respuesta,
   997→      hashtags: '',
   998→      textoPortada: p.pregunta,
   999→      formato: p.modo === 'camara' ? 'hablando a cámara' : 'texto',
  1000→      estado: 'aprobado' as const,
  1001→      fecha: '',
  1002→      diaSemana: '',
  1003→      slides: [],
  1004→      storiesData: [],
  1005→      descripcion: p.pregunta,
  1006→      planId: '',
  1007→      createdAt: new Date().toISOString(),
  1008→    }))
  1009→    addLibraryItems(items)
  1010→    setSavedToLibrary(true)
  1011→    setTimeout(() => setSavedToLibrary(false), 2500)
  1012→  }
  1013→
  1014→  return (
  1015→    <div className="space-y-6">
  1016→      {/* Explanation card */}
  1017→      <Card className="border-[#E8DDD5] bg-[#FFFBF0]">
  1018→        <CardContent className="p-5">
  1019→          <div className="flex items-start gap-3">
  1020→            <div className="w-9 h-9 rounded-full bg-[#FFF1B5] flex items-center justify-center shrink-0">
  1021→              <HelpCircle className="w-4 h-4 text-white" />
  1022→            </div>
  1023→            <div className="flex-1">
