    75→    navigator.clipboard.writeText(text)

    79→export function BancoGanchos() {
    80→  const {
    81→    brandProfile,
    82→    customHooks,
    83→    addCustomHook,
    84→    updateCustomHook,
    85→    removeCustomHook,
    86→    savedHooks,
    87→    saveHook,
    88→    updateSavedHook,
    89→    removeSavedHook,
    90→    addLibraryItem,
    91→    setActiveModule,
    92→  } = useAppStore()
    93→
    94→  const [vista, setVista] = useState<Vista>('banco')
    95→  const [searchTerm, setSearchTerm] = useState('')
    96→  const [categoriaFiltro, setCategoriaFiltro] = useState<string>('')
    97→  const [tipoFiltro, setTipoFiltro] = useState<string>('')
    98→  const [ganchoSeleccionado, setGanchoSeleccionado] = useState<HookCard | null>(null)
    99→  const [generandoGancho, setGenerandoGancho] = useState<string | null>(null)
   100→  const [dialogoGenerar, setDialogoGenerar] = useState<{
   101→    gancho: HookCard
   102→    tipoContenido: 'reel' | 'story' | 'carrusel'
   103→    tono: 'educativo' | 'cercano' | 'vendedor'
   104→    modo: 'texto' | 'camara'
   105→    servicio: string
   106→    resultado: any
   107→    cargando: boolean
   108→    error: string
   109→  } | null>(null)
   110→  const [dialogoEditar, setDialogoEditar] = useState<HookCard | null>(null)
   111→  const [dialogoNuevoGancho, setDialogoNuevoGancho] = useState(false)
   112→  const [copiado, setCopiado] = useState<string | null>(null)
   113→  const [dialogoProgramar, setDialogoProgramar] = useState<SavedHook | null>(null)
   114→
   115→  // Combina ganchos seed + custom
   116→  const todosGanchos = useMemo(() => {
   117→    return [...customHooks, ...HOOKS_SEED]
   118→  }, [customHooks])
   119→
   120→  // Filtrado
   121→  const ganchosFiltrados = useMemo(() => {
   122→    return todosGanchos.filter(g => {
   123→      if (searchTerm) {
   124→        const t = searchTerm.toLowerCase()
   125→        const hit =
   126→          g.titulo.toLowerCase().includes(t) ||
   127→          g.servicio.toLowerCase().includes(t) ||
   128→          g.categoria.toLowerCase().includes(t) ||
   129→          g.explicacion.toLowerCase().includes(t)
   130→        if (!hit) return false
   131→      }
   132→      if (categoriaFiltro && g.categoria !== categoriaFiltro) return false
   133→      if (tipoFiltro && g.tipo !== tipoFiltro) return false
   134→      return true
   135→    })
   136→  }, [todosGanchos, searchTerm, categoriaFiltro, tipoFiltro])
   137→
   138→  const handleCopiar = (id: string, texto: string) => {
   139→    copyToClipboard(texto)
   140→    setCopiado(id)
   141→    setTimeout(() => setCopiado(null), 1500)
   142→  }
   143→
   144→  const handleGuardarIdea = (gancho: HookCard) => {
   145→    const nueva: SavedHook = {
   146→      id: generateId(),
   147→      hookId: gancho.id,
   148→      titulo: gancho.titulo,
   149→      tipoContenido: '',
   150→      servicio: gancho.servicio,
   151→      guionGenerado: '',
   152→      estado: 'idea',
   153→      fechaProgramada: null,
   154→      fechaGrabacion: null,
   155→      createdAt: new Date().toISOString(),
   156→    }
   157→    saveHook(nueva)
   158→    // feedback visual breve
   159→    setCopiado('saved-' + gancho.id)
   160→    setTimeout(() => setCopiado(null), 1500)
   161→  }
   162→
   163→  const yaGuardado = (ganchoId: string) => savedHooks.some(h => h.hookId === ganchoId)
   164→
   165→  const handleGenerar = async () => {
   166→    if (!dialogoGenerar) return
   167→    const { gancho, tipoContenido, tono, modo, servicio } = dialogoGenerar
   168→    setDialogoGenerar({ ...dialogoGenerar, cargando: true, error: '', resultado: null })
   169→    try {
   170→      const res = await fetch('/api/ai', {
   171→        method: 'POST',
   172→        headers: { 'Content-Type': 'application/json' },
   173→        body: JSON.stringify({
   174→          type: 'generar-desde-gancho',
   175→          brandProfile,
   176→          context: {
   177→            gancho: gancho.titulo,
   178→            tipoContenido,
   179→            tono,
   180→            modo,
   181→            servicio: servicio || gancho.servicio,
   182→          },
   183→        }),
   184→      })
   185→      if (!res.ok) throw new Error('Network error')
   186→      const data = await res.json()
   187→      setDialogoGenerar({ ...dialogoGenerar!, cargando: false, resultado: data.result })
   188→    } catch (e: any) {
   189→      setDialogoGenerar({ ...dialogoGenerar!, cargando: false, error: e.message || 'Error generando contenido' })
   190→    }
   191→  }
   192→
   193→  const handleGuardarGenerado = () => {
   194→    if (!dialogoGenerar?.resultado) return
   195→    const { gancho, tipoContenido, servicio, resultado } = dialogoGenerar
   196→    const titulo = resultado.titulo || gancho.titulo
   197→    const guionTexto = tipoContenido === 'carrusel'
   198→      ? (resultado.slides || []).map((s: any) => `Slide ${s.numero}: ${s.texto}`).join('\n\n')
   199→      : tipoContenido === 'story'
   200→      ? (resultado.stories || []).map((s: any) => `Story ${s.numero} (${s.tipo}): ${s.texto}`).join('\n\n')
   201→      : resultado.guion || ''
   202→
   203→    const nueva: SavedHook = {
   204→      id: generateId(),
   205→      hookId: gancho.id,
   206→      titulo,
   207→      tipoContenido,
   208→      servicio: servicio || gancho.servicio,
   209→      guionGenerado: guionTexto + (resultado.copy ? '\n\nCOPY:\n' + resultado.copy : '') + (resultado.hashtags ? '\n\n' + resultado.hashtags : ''),
   210→      estado: 'pendiente',
   211→      fechaProgramada: null,
   212→      fechaGrabacion: null,
   213→      createdAt: new Date().toISOString(),
   214→    }
   215→    saveHook(nueva)
   216→
   217→    // También lo guardamos en la biblioteca general como ContentItem
   218→    addLibraryItem({
   219→      id: generateId(),
   220→      tipo: tipoContenido === 'carrusel' ? 'carrusel' : tipoContenido === 'story' ? 'story' : 'reel',
   221→      titulo,
   222→      objetivo: gancho.objetivo,
   223→      servicio: servicio || gancho.servicio,
   224→      guion: guionTexto,
   225→      copy: resultado.copy || '',
   226→      hashtags: resultado.hashtags || '',
   227→      textoPortada: resultado.textoPortada || '',
   228→      formato: dialogoGenerar.modo === 'camara' ? 'hablando a cámara' : 'texto en pantalla',
   229→      estado: 'borrador',
   230→      fecha: '',
   231→      diaSemana: '',
   232→      slides: resultado.slides || [],
   233→      storiesData: resultado.stories || [],
   234→      descripcion: gancho.titulo,
   235→      planId: '',
   236→      createdAt: new Date().toISOString(),
   237→    })
   238→
   239→    setDialogoGenerar(null)
   240→    setVista('mis-ideas')
   241→  }
   242→
   243→  const abrirDialogoGenerar = (gancho: HookCard) => {
   244→    setGanchoSeleccionado(null)
   245→    setDialogoGenerar({
   246→      gancho,
   247→      tipoContenido: 'reel',
   248→      tono: 'cercano',
   249→      modo: 'camara',
   250→      servicio: gancho.servicio,
   251→      resultado: null,
   252→      cargando: false,
   253→      error: '',
   254→    })
   255→  }
   256→
   257→  return (
   258→    <div className="space-y-6">
   259→      {/* Cabecera */}
   260→      <div className="flex items-start justify-between flex-wrap gap-4">
   261→        <div>
   262→          <h1 className="text-3xl font-bold text-[#2D1F22] mb-1">Banco de Ganchos</h1>
   263→          <p className="text-[#7D2E42]/70">
   264→            Ideas rápidas, llamativas y estratégicas para crear contenido sin quedarte en blanco.
   265→          </p>
   266→        </div>
   267→        <Button
   268→          onClick={() => setDialogoNuevoGancho(true)}
   269→          className="bg-[#7D2E42] hover:bg-[#5d2334] text-white"
   270→        >
   271→          <Plus className="w-4 h-4 mr-1.5" />
   272→          Crear gancho
   273→        </Button>
   274→      </div>
   275→
   276→      {/* Selector de vista */}
   277→      <div className="flex gap-2 bg-white rounded-2xl p-1.5 shadow-sm border border-[#F3E8E5] w-fit">
   278→        {[
   279→          { key: 'banco' as Vista, label: 'Banco de Ganchos', icon: LayoutGrid },
   280→          { key: 'mis-ideas' as Vista, label: 'Mis ideas guardadas', icon: Bookmark },
   281→          { key: 'calendario' as Vista, label: 'Calendario / Lista', icon: Calendar },
   282→        ].map(v => (
   283→          <button
   284→            key={v.key}
   285→            onClick={() => setVista(v.key)}
   286→            className={`flex items-center gap-2 px-4 py-2 rounded-xl text-sm font-medium transition-all ${
   287→              vista === v.key
   288→                ? 'brave-gradient text-white shadow-md'
   289→                : 'text-[#7D2E42] hover:bg-[#FBF7F5]'
   290→            }`}
   291→          >
   292→            <v.icon className="w-4 h-4" />
   293→            {v.label}
   294→          </button>
   295→        ))}
   296→      </div>
   297→
   298→      {/* VISTA: BANCO */}
   299→      {vista === 'banco' && (
   300→        <>
   301→          {/* Filtros */}
   302→          <Card className="border-[#F3E8E5] shadow-sm">
   303→            <CardContent className="p-4 space-y-3">
   304→              <div className="relative">
   305→                <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-[#7D2E42]/50" />
   306→                <Input
   307→                  value={searchTerm}
   308→                  onChange={(e) => setSearchTerm(e.target.value)}
   309→                  placeholder="Buscar ganchos por título, servicio, categoría..."
   310→                  className="pl-10 bg-white border-[#F3E8E5]"
   311→                />
   312→              </div>
   313→              <div className="space-y-2">
   314→                <p className="text-xs font-semibold uppercase tracking-wider text-[#7D2E42]/60 flex items-center gap-1.5">
   315→                  <Filter className="w-3 h-3" /> Categoría
   316→                </p>
   317→                <div className="flex flex-wrap gap-1.5">
   318→                  <button
   319→                    onClick={() => setCategoriaFiltro('')}
   320→                    className={`text-xs px-3 py-1 rounded-full border transition-colors ${
   321→                      !categoriaFiltro
   322→                        ? 'bg-[#7D2E42] text-white border-[#7D2E42]'
   323→                        : 'bg-white text-[#7D2E42] border-[#F3E8E5] hover:bg-[#FBF7F5]'
   324→                    }`}
   325→                  >
   326→                    Todas
   327→                  </button>
   328→                  {HOOK_CATEGORIAS.map(c => (
   329→                    <button
   330→                      key={c}
   331→                      onClick={() => setCategoriaFiltro(c === categoriaFiltro ? '' : c)}
   332→                      className={`text-xs px-3 py-1 rounded-full border transition-colors ${
   333→                        c === categoriaFiltro
   334→                          ? 'bg-[#7D2E42] text-white border-[#7D2E42]'
   335→                          : 'bg-white text-[#7D2E42] border-[#F3E8E5] hover:bg-[#FBF7F5]'
   336→                      }`}
   337→                    >
   338→                      {c}
   339→                    </button>
   340→                  ))}
   341→                </div>
   342→              </div>
   343→              <div className="space-y-2">
   344→                <p className="text-xs font-semibold uppercase tracking-wider text-[#7D2E42]/60 flex items-center gap-1.5">
   345→                  <Zap className="w-3 h-3" /> Tipo de gancho
   346→                </p>
   347→                <div className="flex flex-wrap gap-1.5">
   348→                  <button
   349→                    onClick={() => setTipoFiltro('')}
   350→                    className={`text-xs px-3 py-1 rounded-full border transition-colors ${
   351→                      !tipoFiltro
   352→                        ? 'bg-[#C9A96E] text-[#2D1F22] border-[#C9A96E]'
   353→                        : 'bg-white text-[#7D2E42] border-[#F3E8E5] hover:bg-[#FBF7F5]'
   354→                    }`}
   355→                  >
   356→                    Todos
   357→                  </button>
   358→                  {HOOK_TIPOS.map(t => (
   359→                    <button
   360→                      key={t}
   361→                      onClick={() => setTipoFiltro(t === tipoFiltro ? '' : t)}
   362→                      className={`text-xs px-3 py-1 rounded-full border transition-colors ${
   363→                        t === tipoFiltro
   364→                          ? 'bg-[#C9A96E] text-[#2D1F22] border-[#C9A96E]'
   365→                          : 'bg-white text-[#7D2E42] border-[#F3E8E5] hover:bg-[#FBF7F5]'
   366→                      }`}
   367→                    >
   368→                      {t}
   369→                    </button>
   370→                  ))}
   371→                </div>
   372→              </div>
   373→              {(categoriaFiltro || tipoFiltro || searchTerm) && (
   374→                <div className="flex items-center gap-2 text-xs text-[#7D2E42]/70 pt-2 border-t border-[#F3E8E5]">
   375→                  <span>{ganchosFiltrados.length} ganchos encontrados</span>
   376→                  <button
   377→                    onClick={() => {
   378→                      setCategoriaFiltro('')
   379→                      setTipoFiltro('')
   380→                      setSearchTerm('')
   381→                    }}
   382→                    className="text-[#C17C83] hover:underline ml-auto"
   383→                  >
   384→                    Limpiar filtros
   385→                  </button>
   386→                </div>
   387→              )}
   388→            </CardContent>
   389→          </Card>
   390→
   391→          {/* Grid de ganchos */}
   392→          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   393→            {ganchosFiltrados.map(gancho => {
   394→              const guardado = yaGuardado(gancho.id)
   395→              return (
   396→                <Card
   397→                  key={gancho.id}
   398→                  className="border-[#F3E8E5] shadow-sm hover:shadow-md transition-all flex flex-col"
   399→                >
   400→                  <CardHeader className="pb-3">
   401→                    <div className="flex items-start justify-between gap-2 mb-2">
   402→                      <span className={`text-[10px] font-bold uppercase tracking-wider px-2 py-0.5 rounded ${IMPACTO_COLOR[gancho.impacto] || IMPACTO_COLOR.Medio}`}>
   403→                        {gancho.impacto}
   404→                      </span>
   405→                      <span className={`text-[10px] px-2 py-0.5 rounded border ${TIPO_COLOR[gancho.tipo] || 'bg-gray-100 text-gray-700 border-gray-200'}`}>
   406→                        {gancho.tipo}
   407→                      </span>
   408→                    </div>
   409→                    <CardTitle className="text-base text-[#2D1F22] leading-snug">
   410→                      {gancho.titulo}
   411→                    </CardTitle>
   412→                    <CardDescription className="flex flex-wrap gap-1.5 mt-2">
   413→                      <span className="text-[10px] px-2 py-0.5 rounded-full bg-[#FBF7F5] border border-[#F3E8E5] text-[#7D2E42]">
   414→                        {gancho.categoria}
   415→                      </span>
   416→                      <span className="text-[10px] px-2 py-0.5 rounded-full bg-[#FBF7F5] border border-[#F3E8E5] text-[#7D2E42]">
   417→                        {gancho.servicio}
   418→                      </span>
   419→                      <span className="text-[10px] px-2 py-0.5 rounded-full bg-[#FBF7F5] border border-[#F3E8E5] text-[#7D2E42]">
   420→                        Objetivo: {gancho.objetivo}
   421→                      </span>
   422→                    </CardDescription>
   423→                  </CardHeader>
   424→                  <CardContent className="pt-0 flex-1 flex flex-col">
   425→                    <p className="text-xs text-[#2D1F22]/70 line-clamp-2 mb-3 flex-1">
   426→                      {gancho.explicacion}
   427→                    </p>
   428→                    <div className="flex flex-wrap gap-1.5">
   429→                      <Button
   430→                        size="sm"
   431→                        onClick={() => abrirDialogoGenerar(gancho)}
   432→                        className="flex-1 h-8 text-xs brave-gradient hover:opacity-90"
   433→                      >
   434→                        <Sparkles className="w-3 h-3 mr-1" />
   435→                        Generar guion
   436→                      </Button>
   437→                      <Button
   438→                        size="sm"
   439→                        variant="outline"
   440→                        onClick={() => setGanchoSeleccionado(gancho)}
   441→                        className="h-8 text-xs border-[#F3E8E5] text-[#7D2E42] hover:bg-[#FBF7F5]"
   442→                      >
   443→                        Ver detalle
   444→                      </Button>
   445→                    </div>
   446→                    <div className="flex items-center gap-1.5 mt-2">
   447→                      <Button
   448→                        size="sm"
   449→                        variant="ghost"
   450→                        onClick={() => handleGuardarIdea(gancho)}
   451→                        disabled={guardado}
   452→                        className="flex-1 h-7 text-[11px] text-[#7D2E42] hover:bg-[#FBF7F5]"
   453→                      >
   454→                        {guardado ? (
   455→                          <><Check className="w-3 h-3 mr-1" /> Guardado</>
   456→                        ) : copiado === 'saved-' + gancho.id ? (
   457→                          <><Check className="w-3 h-3 mr-1 text-green-600" /> Guardado</>
   458→                        ) : (
   459→                          <><Bookmark className="w-3 h-3 mr-1" /> Guardar idea</>
   460→                        )}
   461→                      </Button>
   462→                      <Button
   463→                        size="sm"
   464→                        variant="ghost"
   465→                        onClick={() => handleCopiar(gancho.id, gancho.titulo)}
   466→                        className="h-7 text-[11px] text-[#7D2E42] hover:bg-[#FBF7F5] px-2"
   467→                      >
   468→                        {copiado === gancho.id ? <Check className="w-3 h-3 text-green-600" /> : <Copy className="w-3 h-3" />}
   469→                      </Button>
   470→                      {gancho.esCustom && (
   471→                        <>
   472→                          <Button
   473→                            size="sm"
   474→                            variant="ghost"
   475→                            onClick={() => setDialogoEditar(gancho)}
   476→                            className="h-7 text-[11px] text-[#7D2E42] hover:bg-[#FBF7F5] px-2"
   477→                          >
   478→                            <Edit2 className="w-3 h-3" />
   479→                          </Button>
   480→                          <Button
   481→                            size="sm"
   482→                            variant="ghost"
   483→                            onClick={() => {
   484→                              if (confirm('¿Eliminar este gancho personalizado?')) {
   485→                                removeCustomHook(gancho.id)
   486→                              }
   487→                            }}
   488→                            className="h-7 text-[11px] text-[#C17C83] hover:bg-[#C17C83]/10 px-2"
   489→                          >
   490→                            <Trash2 className="w-3 h-3" />
   491→                          </Button>
   492→                        </>
   493→                      )}

   513→      {/* VISTA: MIS IDEAS GUARDADAS */}
   514→      {vista === 'mis-ideas' && (

   585→      )}
   586→
   587→      {/* DIÁLOGO: Programar fecha */}
   588→      {dialogoProgramar && (

   598→            setDialogoProgramar(null)

   601→      )}
   602→    </div>
   603→  )
   604→}

  1352→function CalendarioSimple({
  1353→  savedHooks,
  1354→  onUpdate,
  1355→}: {
  1356→  savedHooks: SavedHook[]
  1357→  onUpdate: (id: string, updates: Partial<SavedHook>) => void
  1358→}) {
  1359→  const [mes, setMes] = useState(() => {
  1360→    const d = new Date()
  1361→    return new Date(d.getFullYear(), d.getMonth(), 1)
  1362→  })
  1363→  const [vistaLista, setVistaLista] = useState(false)
  1364→
  1365→  // Hooks con fecha programada
  1366→  const programados = savedHooks.filter(h => h.fechaProgramada)
  1367→  const porFecha = new Map<string, SavedHook[]>()
  1368→  programados.forEach(h => {
  1369→    if (!h.fechaProgramada) return
  1370→    const arr = porFecha.get(h.fechaProgramada) || []
  1371→    arr.push(h)
  1372→    porFecha.set(h.fechaProgramada, arr)
  1373→  })
  1374→
  1375→  // Generar días del mes
  1376→  const inicioMes = new Date(mes.getFullYear(), mes.getMonth(), 1)
  1377→  const finMes = new Date(mes.getFullYear(), mes.getMonth() + 1, 0)
  1378→  const inicioGrid = new Date(inicioMes)
  1379→  inicioGrid.setDate(inicioGrid.getDate() - inicioGrid.getDay()) // Domingo = 0
  1380→  const dias: Date[] = []
  1381→  const cursor = new Date(inicioGrid)
  1382→  while (cursor <= finMes || dias.length % 7 !== 0) {
  1383→    dias.push(new Date(cursor))
  1384→    cursor.setDate(cursor.getDate() + 1)
  1385→    if (dias.length > 42) break
  1386→  }
  1387→
  1388→  const monthName = mes.toLocaleDateString('es-ES', { month: 'long', year: 'numeric' })
  1389→
  1390→  return (
  1391→    <div className="space-y-4">
  1392→      <div className="flex items-center justify-between flex-wrap gap-3">
  1393→        <div className="flex items-center gap-2 bg-white rounded-xl p-1 border border-[#F3E8E5]">
  1394→          <button
  1395→            onClick={() => setVistaLista(false)}
  1396→            className={`text-xs px-3 py-1.5 rounded-lg ${!vistaLista ? 'bg-[#7D2E42] text-white' : 'text-[#7D2E42]'}`}
  1397→          >
  1398→            Calendario
  1399→          </button>
  1400→          <button
  1401→            onClick={() => setVistaLista(true)}
  1402→            className={`text-xs px-3 py-1.5 rounded-lg ${vistaLista ? 'bg-[#7D2E42] text-white' : 'text-[#7D2E42]'}`}
  1403→          >
  1404→            Lista
  1405→          </button>
  1406→        </div>
  1407→        {!vistaLista && (
  1408→          <div className="flex items-center gap-2">
  1409→            <Button
  1410→              size="sm"
  1411→              variant="outline"
  1412→              onClick={() => setMes(new Date(mes.getFullYear(), mes.getMonth() - 1, 1))}
  1413→              className="border-[#F3E8E5] text-[#7D2E42]"
  1414→            >
  1415→              ←
  1416→            </Button>
  1417→            <span className="text-sm font-semibold capitalize text-[#2D1F22] min-w-[140px] text-center">{monthName}</span>
  1418→            <Button
  1419→              size="sm"
