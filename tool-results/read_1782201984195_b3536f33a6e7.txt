    69→// ============================================================
    70→export function BancoGanchos() {
    71→  const { brandProfile, savedHooks, saveHook, updateSavedHook, removeSavedHook, addLibraryItem, setIsLoading } = useAppStore()
    72→
    73→  const [vista, setVista] = useState<Vista>('banco')
    74→  const [categoriaActiva, setCategoriaActiva] = useState('')
    75→  const [searchTerm, setSearchTerm] = useState('')
    76→  const [ganchos, setGanchos] = useState<HookCard[]>([])
    77→  const [generando, setGenerando] = useState(false)
    78→  const [copiado, setCopiado] = useState<string | null>(null)
    79→
    80→  // Modal state
    81→  const [hookSeleccionado, setHookSeleccionado] = useState<HookCard | null>(null)
    82→  const [modalOpen, setModalOpen] = useState(false)
    83→  const [resultadoGenerado, setResultadoGenerado] = useState<any>(null)
    84→  const [generandoScript, setGenerandoScript] = useState(false)
    85→  const [tip, setTip] = useState(getRandomMotivationalTip())
    86→
    87→  // Ref to avoid generating on first render
    88→  const hasInitialized = useRef(false)
    89→
    90→  // Auto-generate on mount
    91→  useEffect(() => {
    92→    if (!hasInitialized.current) {
    93→      hasInitialized.current = true
    94→      generarGanchos('')
    95→    }
    96→  }, [])
    97→
    98→  // Generate hooks via AI
    99→  const generarGanchos = useCallback(async (categoria: string, search?: string) => {
   100→    setGenerando(true)
   101→    setTip(getRandomMotivationalTip())
   102→    try {
   103→      const res = await fetch('/api/ai', {
   104→        method: 'POST',
   105→        headers: { 'Content-Type': 'application/json' },
   106→        body: JSON.stringify({
   107→          type: 'ganchos-extra',
   108→          brandProfile,
   109→          context: {
   110→            categoria: categoria || undefined,
   111→            tipo: undefined,
   112→            numGanchos: 12,
   113→          },
   114→        }),
   115→      })
   116→      const data = await res.json()
   117→      if (data.result?.ganchos && Array.isArray(data.result.ganchos)) {
   118→        const parsed = data.result.ganchos.map((g: any, i: number) => ({
   119→          id: `hook-gen-${Date.now()}-${i}`,
   120→          titulo: g.titulo || '',
   121→          categoria: g.categoria || categoria || 'General',
   122→          tipo: g.tipo || 'Viral',
   123→          objetivo: g.objetivo || 'visibilidad',
   124→          servicio: g.servicio || '',
   125→          impacto: g.impacto || 'Medio',
   126→          explicacion: g.explicacion || '',
   127→          dolor: g.dolor || '',
   128→          deseo: g.deseo || '',
   129→          ideaVisual: g.ideaVisual || '',
   130→        }))
   131→        // If search term, filter locally
   132→        if (search) {
   133→          const q = search.toLowerCase()
   134→          const filtered = parsed.filter((g: HookCard) =>
   135→            g.titulo.toLowerCase().includes(q) ||
   136→            g.categoria.toLowerCase().includes(q) ||
   137→            g.servicio.toLowerCase().includes(q) ||
   138→            g.explicacion.toLowerCase().includes(q)
   139→          )
   140→          setGanchos(filtered.length > 0 ? filtered : parsed)
   141→        } else {
   142→          setGanchos(parsed)
   143→        }
   144→      } else {
   145→        // Fallback: generate basic hooks
   146→        setGanchos(generateFallbackHooks(categoria))
   147→      }
   148→    } catch {
   149→      setGanchos(generateFallbackHooks(categoria))
   150→    } finally {
   151→      setGenerando(false)
   152→    }
   153→  }, [brandProfile])
   154→
   155→  // Handle category click
   156→  const handleCategoria = (cat: string) => {
   157→    const nueva = cat === categoriaActiva ? '' : cat
   158→    setCategoriaActiva(nueva)
   159→    generarGanchos(nueva, searchTerm || undefined)
   160→  }
   161→
   162→  // Handle search
   163→  const handleSearch = (term: string) => {
   164→    setSearchTerm(term)
   165→    // Debounce search
   166→    if (term.length >= 2) {
   167→      const timer = setTimeout(() => generarGanchos(categoriaActiva, term), 600)
   168→      return () => clearTimeout(timer)
   169→    } else if (term.length === 0) {
   170→      generarGanchos(categoriaActiva)
   171→    }
   172→  }
   173→
   174→  // Regenerate
   175→  const handleRegenerar = () => {
   176→    generarGanchos(categoriaActiva, searchTerm || undefined)
   177→  }
   178→
   179→  const yaGuardado = (hookId: string) => savedHooks.some(h => h.hookId === hookId)
   180→
   181→  // Save hook to library
   182→  const handleGuardar = (gancho: HookCard) => {
   183→    if (yaGuardado(gancho.id)) return
   184→    const nueva: SavedHook = {
   185→      id: generateId(),
   186→      hookId: gancho.id,
   187→      titulo: gancho.titulo,
   188→      tipoContenido: '',
   189→      servicio: gancho.servicio,
   190→      guionGenerado: '',
   191→      estado: 'idea',
   192→      fechaProgramada: null,
   193→      fechaGrabacion: null,
   194→      createdAt: new Date().toISOString(),
   195→    }
   196→    saveHook(nueva)
   197→    setCopiado('saved-' + gancho.id)
   198→    setTimeout(() => setCopiado(null), 1800)
   199→  }
   200→
   201→  // Open hook to read full script
   202→  const abrirHook = (gancho: HookCard) => {
   203→    setHookSeleccionado(gancho)
   204→    setResultadoGenerado(null)
   205→    setModalOpen(true)
   206→  }
   207→
   208→  // Generate content from hook
   209→  const handleGenerar = async () => {
   210→    if (!hookSeleccionado) return
   211→    setGenerandoScript(true)
   212→    setIsLoading(true, 'Generando guion completo...')
   213→    try {
   214→      const res = await fetch('/api/ai', {
   215→        method: 'POST',
   216→        headers: { 'Content-Type': 'application/json' },
   217→        body: JSON.stringify({
   218→          type: 'generar-desde-gancho',
   219→          brandProfile,
   220→          context: {
   221→            gancho: hookSeleccionado.titulo,
   222→            tipoContenido: 'reel',
   223→            tono: 'cercano',
   224→            modo: 'camara',
   225→            servicio: hookSeleccionado.servicio,
   226→          },
   227→        }),
   228→      })
   229→      if (!res.ok) throw new Error('Network error')
   230→      const data = await res.json()
   231→      setResultadoGenerado(data.result)
   232→    } catch (e) {
   233→      console.error('Generate error:', e)
   234→    } finally {
   235→      setGenerandoScript(false)
   236→      setIsLoading(false)
   237→    }
   238→  }
   239→
   240→  // Save generated content to library
   241→  const handleGuardarGenerado = () => {
   242→    if (!hookSeleccionado || !resultadoGenerado) return
   243→    const titulo = resultadoGenerado.titulo || hookSeleccionado.titulo
   244→    const guionTexto = resultadoGenerado.guion || ''
   245→
   246→    saveHook({
   247→      id: generateId(),
   248→      hookId: hookSeleccionado.id,
   249→      titulo,
   250→      tipoContenido: 'reel',
   251→      servicio: hookSeleccionado.servicio,
   252→      guionGenerado: guionTexto + (resultadoGenerado.copy ? '\n\nCOPY:\n' + resultadoGenerado.copy : '') + (resultadoGenerado.hashtags ? '\n\n' + resultadoGenerado.hashtags : ''),
   253→      estado: 'pendiente',
   254→      fechaProgramada: null,
   255→      fechaGrabacion: null,
   256→      createdAt: new Date().toISOString(),
   257→    })
   258→
   259→    addLibraryItem({
   260→      id: generateId(),
   261→      tipo: 'reel',
   262→      titulo,
   263→      objetivo: hookSeleccionado.objetivo,
   264→      servicio: hookSeleccionado.servicio,
   265→      guion: guionTexto,
   266→      copy: resultadoGenerado.copy || '',
   267→      hashtags: resultadoGenerado.hashtags || '',
   268→      textoPortada: resultadoGenerado.textoPortada || '',
   269→      formato: 'hablando a cámara',
   270→      estado: 'borrador',
   271→      fecha: '',
   272→      diaSemana: '',
   273→      slides: resultadoGenerado.slides || [],
   274→      storiesData: resultadoGenerado.stories || [],
   275→      descripcion: hookSeleccionado.titulo,
   276→      planId: '',
   277→      createdAt: new Date().toISOString(),
   278→    })
   279→
   280→    setCopiado('generado')
   281→    setTimeout(() => setCopiado(null), 1800)
   282→  }
   283→
   284→  return (
   285→    <div className="space-y-5">
   286→      {/* Header */}
   287→      <div className="flex items-start justify-between gap-4">
   288→        <div>
   289→          <h1 className="text-2xl font-bold text-foreground tracking-tight">Banco de Ganchos</h1>
   290→          <p className="text-sm text-muted-foreground mt-0.5">Ganchos que se regeneran según lo que buscas</p>
   291→        </div>
   292→        <Button
   293→          onClick={handleRegenerar}
   294→          disabled={generando}
   295→          className="shrink-0 h-10 px-4 rounded-xl brave-gradient hover:opacity-90 text-white font-medium text-sm gap-2 shadow-md shadow-brave-chocolate/15"
   296→        >
   297→          {generando ? (
   298→            <Loader2 className="w-4 h-4 animate-spin" />
   299→          ) : (
   300→            <RefreshCw className="w-4 h-4" />
   301→          )}
   302→          Regenerar
   303→        </Button>
   304→      </div>
   305→
   306→      {/* View toggle */}
   307→      <div className="flex gap-1 bg-card rounded-2xl p-1 shadow-sm border border-border w-fit">
   308→        {[
   309→          { key: 'banco' as Vista, label: 'Explorar', icon: Lightbulb },
   310→          { key: 'guardados' as Vista, label: 'Guardados', icon: Bookmark },
   311→        ].map(v => (
   312→          <button
   313→            key={v.key}
   314→            onClick={() => setVista(v.key)}
   315→            className={`flex items-center gap-1.5 px-4 py-2 rounded-xl text-sm font-medium transition-all ${
   316→              vista === v.key
   317→                ? 'brave-gradient text-white shadow-md'
   318→                : 'text-muted-foreground hover:text-foreground hover:bg-muted/50'
   319→            }`}
   320→          >
   321→            <v.icon className="w-4 h-4" />
   322→            {v.label}
   323→            {v.key === 'guardados' && savedHooks.length > 0 && (
   324→              <span className={`text-[10px] px-1.5 py-0.5 rounded-full ${
   325→                vista === v.key ? 'bg-white/25' : 'bg-muted'
   326→              }`}>
   327→                {savedHooks.length}
   328→              </span>
   329→            )}
   330→          </button>
   331→        ))}
   332→      </div>
   333→
   334→      {/* VISTA: EXPLORAR */}
   335→      {vista === 'banco' && (
   336→        <div className="space-y-4">
   337→          {/* Mascot motivational tip */}
   338→          <motion.div
   339→            key={tip}
   340→            initial={{ opacity: 0, y: 4 }}
   341→            animate={{ opacity: 1, y: 0 }}
   342→            className="mascot-banner rounded-2xl px-4 py-3 flex items-center gap-3"
   343→          >
   344→            <div className="brave-bounce">
   345→              <BravyBot size={38} expression="wink" animate />
   346→            </div>
   347→            <p className="text-xs font-medium text-foreground leading-snug flex-1">{tip}</p>
   348→          </motion.div>
   349→
   350→          {/* Search */}
   351→          <div className="relative">
   352→            <Search className="absolute left-3.5 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   353→            <Input
   354→              value={searchTerm}
   355→              onChange={(e) => handleSearch(e.target.value)}
   356→              placeholder="Buscar tema, servicio o palabra clave..."
   357→              className="pl-10 bg-card border-border rounded-xl h-11 text-sm"
   358→            />
   359→            {searchTerm && (
   360→              <button
   361→                onClick={() => { setSearchTerm(''); generarGanchos(categoriaActiva) }}
   362→                className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
   363→              >
   364→                <X className="w-4 h-4" />
   365→              </button>
   366→            )}
   367→          </div>
   368→
   369→          {/* Category chips */}
   370→          <div className="flex gap-2 overflow-x-auto pb-1 -mx-1 px-1 no-scrollbar">
   371→            <button
   372→              onClick={() => handleCategoria('')}
   373→              className={`brave-chip shrink-0 border ${
   374→                !categoriaActiva
   375→                  ? 'bg-foreground text-background border-foreground'
   376→                  : 'bg-card text-muted-foreground border-border'
   377→              }`}
   378→            >
   379→              Todos
   380→            </button>
   381→            {CATEGORIAS.map(cat => {
   382→              const isActive = categoriaActiva === cat.nombre
   383→              return (
   384→                <button
   385→                  key={cat.nombre}
   386→                  onClick={() => handleCategoria(cat.nombre)}
   387→                  className={`brave-chip shrink-0 border ${isActive ? cat.active : cat.color}`}
   388→                >
   389→                  {cat.nombre}
   390→                </button>
   391→              )
   392→            })}
   393→          </div>
   394→
   395→          {/* Generating state */}
   396→          {generando ? (
   397→            <div className="flex flex-col items-center justify-center py-16">
   398→              <div className="brave-float">
   399→                <BravyBot size={72} expression="excited" animate speechBubble="Creando ganchos..." />
   400→              </div>
   401→              <div className="mt-5 w-48 h-2 bg-muted rounded-full overflow-hidden">
   402→                <div className="h-full brave-shimmer rounded-full" />
   403→              </div>
   404→              <p className="text-sm text-muted-foreground mt-3 font-medium">
   405→                {categoriaActiva ? `Buscando los mejores ganchos de ${categoriaActiva}...` : 'Generando ganchos para ti...'}
   406→              </p>
   407→            </div>
   408→          ) : (
   409→            <>
   410→              {/* Results count */}
   411→              <div className="flex items-center justify-between">
   412→                <p className="text-xs text-muted-foreground font-medium">
   413→                  {ganchos.length} ganchos {categoriaActiva && `en ${categoriaActiva}`}
   414→                </p>
   415→                <button
   416→                  onClick={handleRegenerar}
   417→                  className="text-xs text-muted-foreground hover:text-foreground font-medium flex items-center gap-1 transition-colors"
   418→                >
   419→                  <Zap className="w-3 h-3" /> Más opciones
   420→                </button>
   421→              </div>
   422→
   423→              {/* Hook cards grid */}
   424→              <div className="grid grid-cols-1 md:grid-cols-2 xl:grid-cols-3 gap-3">
   425→                <AnimatePresence mode="popLayout">
   426→                  {ganchos.map((gancho, idx) => {
   427→                    const guardado = yaGuardado(gancho.id)
   428→                    const catInfo = CATEGORIA_MAP[gancho.categoria]
   429→                    return (
   430→                      <motion.div
   431→                        key={gancho.id}
   432→                        initial={{ opacity: 0, y: 16, scale: 0.97 }}
   433→                        animate={{ opacity: 1, y: 0, scale: 1 }}
   434→                        exit={{ opacity: 0, y: -8, scale: 0.97 }}
   435→                        transition={{ duration: 0.3, delay: Math.min(idx * 0.04, 0.4) }}
   436→                      >
   437→                        <Card className="bg-card border-border/70 shadow-sm hover:shadow-lg hover:-translate-y-1 transition-all duration-300 rounded-2xl overflow-hidden group">
   438→                          {/* Colored top bar */}
   439→                          <div className={`h-1 ${catInfo?.active?.replace('text-white', '').replace('border-', 'bg-').split(' ')[0] || 'bg-brave-pastel-blue'}`} />
   440→
   441→                          <CardContent className="p-4 space-y-3">
   442→                            {/* Category + Impact */}
   443→                            <div className="flex items-center justify-between">
   444→                              <span className={`text-[10px] font-semibold px-2.5 py-0.5 rounded-full border ${catInfo?.color || 'bg-muted text-muted-foreground border-border'}`}>
   445→                                {gancho.categoria}
   446→                              </span>
   447→                              <span className={`text-[10px] font-bold px-2 py-0.5 rounded-full ${
   448→                                gancho.impacto === 'Alto'
   449→                                  ? 'bg-red-50 text-red-600'
   450→                                  : gancho.impacto === 'Medio'
   451→                                  ? 'bg-amber-50 text-amber-600'
   452→                                  : 'bg-slate-50 text-slate-500'
   453→                              }`}>
   454→                                {gancho.impacto}
   455→                              </span>
   456→                            </div>
   457→
   458→                            {/* Hook text - the star of the card */}
   459→                            <p className="text-[15px] font-bold text-foreground leading-snug group-hover:text-primary transition-colors">
   460→                              &ldquo;{gancho.titulo}&rdquo;
   461→                            </p>
   462→
   463→                            {/* Brief explanation */}
   464→                            <p className="text-xs text-muted-foreground leading-relaxed line-clamp-2">
   465→                              {gancho.explicacion}
   466→                            </p>
   467→
   468→                            {/* 3 ACTION BUTTONS */}
   469→                            <div className="flex gap-2 pt-1">
   470→                              {/* 1. Guardar en biblioteca */}
   471→                              <button
   472→                                onClick={() => handleGuardar(gancho)}
   473→                                disabled={guardado}
   474→                                className={`flex-1 flex items-center justify-center gap-1.5 h-9 rounded-xl text-xs font-semibold transition-all ${
   475→                                  guardado
   476→                                    ? 'bg-muted/70 text-muted-foreground cursor-default'
   477→                                    : 'bg-card border border-border text-foreground hover:bg-brave-pastel-blue/30 hover:border-brave-pastel-blue'
   478→                                }`}
   479→                              >
   480→                                {copiado === 'saved-' + gancho.id ? (
   481→                                  <><Check className="w-3.5 h-3.5 text-emerald-500" /> Guardado</>
   482→                                ) : guardado ? (
   483→                                  <><Bookmark className="w-3.5 h-3.5" /> En biblioteca</>
   484→                                ) : (
   485→                                  <><Bookmark className="w-3.5 h-3.5" /> Guardar</>
   486→                                )}
   487→                              </button>
   488→
   489→                              {/* 2. Abrir y leer guion entero */}
   490→                              <button
   491→                                onClick={() => abrirHook(gancho)}
   492→                                className="flex-1 flex items-center justify-center gap-1.5 h-9 rounded-xl text-xs font-semibold brave-gradient text-white hover:opacity-90 transition-all shadow-sm shadow-brave-chocolate/10"
   493→                              >
   494→                                <FileText className="w-3.5 h-3.5" />
   495→                                Guion
   496→                                <ArrowRight className="w-3 h-3" />
